40 WorkflowSpec defineDataProcessing(ConfigContext const& config)
41 {
42  using namespace o2::tpc;
43 
44  // set up configuration
45  o2::conf::ConfigurableParam::updateFromString(config.options().get<std::string>("configKeyValues"));
46  o2::conf::ConfigurableParam::writeINI("o2tpcdistributeidc_configuration.ini");
47 
48  const auto tpcCRUs = o2::RangeTokenizer::tokenize<int>(config.options().get<std::string>("crus"));
49  const auto nCRUs = tpcCRUs.size();
50  const auto timeframes = static_cast<unsigned int>(config.options().get<int>("timeframes"));
51  const auto outlanes = static_cast<unsigned int>(config.options().get<int>("output-lanes"));
52  const auto firstTF = static_cast<unsigned int>(config.options().get<int>("firstTF"));
53  const auto loadFromFile = config.options().get<bool>("load-from-file");
54 
55  const auto first = tpcCRUs.begin();
56  const auto last = std::min(tpcCRUs.end(), first + nCRUs);
57  const std::vector<uint32_t> rangeCRUs(first, last);
58  WorkflowSpec workflow{getTPCDistributeIDCSpec(rangeCRUs, timeframes, outlanes, firstTF, loadFromFile)};
59  return workflow;
60 }
