46 template <typename T>
47 void Tracker<T>::initConfig(const MFTTrackingParam& trkParam, bool printConfig)
48 {
50 
51  mTrackFitter->setMFTRadLength(trkParam.MFTRadLength);
52  mTrackFitter->setVerbosity(trkParam.verbose);
53  mTrackFitter->setTrackModel(trkParam.trackmodel);
54 
55  mMinTrackPointsLTF = trkParam.MinTrackPointsLTF;
56  mMinTrackPointsCA = trkParam.MinTrackPointsCA;
57  mMinTrackStationsLTF = trkParam.MinTrackStationsLTF;
58  mMinTrackStationsCA = trkParam.MinTrackStationsCA;
59  mLTFclsRCut = trkParam.LTFclsRCut;
60  mLTFclsR2Cut = mLTFclsRCut * mLTFclsRCut;
61  mROADclsRCut = trkParam.ROADclsRCut;
62  mROADclsR2Cut = mROADclsRCut * mROADclsRCut;
63  mLTFseed2BinWin = trkParam.LTFseed2BinWin;
64  mLTFinterBinWin = trkParam.LTFinterBinWin;
65  mLTFConeRadius = trkParam.LTFConeRadius;
66  mCAConeRadius = trkParam.CAConeRadius;
67 
68  mRBins = trkParam.RBins;
69  mPhiBins = trkParam.PhiBins;
70  mRPhiBins = trkParam.RBins * trkParam.PhiBins;
71  if (mRPhiBins > constants::index_table::MaxRPhiBins) {
72  LOG(warn) << "To many RPhiBins for this configuration!";
73  mRPhiBins = constants::index_table::MaxRPhiBins;
74  mRBins = sqrt(constants::index_table::MaxRPhiBins);
75  mPhiBins = sqrt(constants::index_table::MaxRPhiBins);
76  LOG(warn) << "Using instead RBins " << mRBins << " and PhiBins " << mPhiBins;
77  }
78  mRBinSize = (constants::index_table::RMax - constants::index_table::RMin) / mRBins;
79  mPhiBinSize = (constants::index_table::PhiMax - constants::index_table::PhiMin) / mPhiBins;
80 
81  if (printConfig) {
82  LOG(info) << "Configurable tracker parameters:";
83  LOG(info) << "MinTrackPointsLTF = " << mMinTrackPointsLTF;
84  LOG(info) << "MinTrackPointsCA = " << mMinTrackPointsCA;
85  LOG(info) << "MinTrackStationsLTF = " << mMinTrackStationsLTF;
86  LOG(info) << "MinTrackStationsCA = " << mMinTrackStationsCA;
87  LOG(info) << "LTFclsRCut = " << mLTFclsRCut;
88  LOG(info) << "ROADclsRCut = " << mROADclsRCut;
89  LOG(info) << "RBins = " << mRBins;
90  LOG(info) << "PhiBins = " << mPhiBins;
91  LOG(info) << "LTFseed2BinWin = " << mLTFseed2BinWin;
92  LOG(info) << "LTFinterBinWin = " << mLTFinterBinWin;
93  LOG(info) << "FullClusterScan = " << (trkParam.FullClusterScan ? "true" : "false");
94  LOG(info) << "forceZeroField = " << (trkParam.forceZeroField ? "true" : "false");
95  }
96 }

99 template <typename T>
100 void Tracker<T>::initialize(bool fullClusterScan)
101 {
102  mRoad.initialize();
103 
104  if (fullClusterScan) {
105  mFullClusterScan = true;
106  return;
107  }
108 
111 
112  Float_t dz, x, y, r, phi, x_proj, y_proj, r_proj, phi_proj;
113  Int_t binIndex1, binIndex2, binIndex2S, binR_proj, binPhi_proj;
114 
115  for (Int_t layer1 = 0; layer1 < (constants::mft::LayersNumber - 1); ++layer1) {
116 
117  for (Int_t iRBin = 0; iRBin < mRBins; ++iRBin) {
118 
119  r = (iRBin + 0.5) * mRBinSize + constants::index_table::RMin;
120 
121  for (Int_t iPhiBin = 0; iPhiBin < mPhiBins; ++iPhiBin) {
122 
123  phi = (iPhiBin + 0.5) * mPhiBinSize + constants::index_table::PhiMin;
124 
125  binIndex1 = getBinIndex(iRBin, iPhiBin);
126 
127  x = r * TMath::Cos(phi);
128  y = r * TMath::Sin(phi);
129 
130  for (Int_t layer2 = (layer1 + 1); layer2 < constants::mft::LayersNumber; ++layer2) {
131 
132  dz = constants::mft::LayerZCoordinate()[layer2] - constants::mft::LayerZCoordinate()[layer1];
133  x_proj = x + dz * x * constants::mft::InverseLayerZCoordinate()[layer1];
134  y_proj = y + dz * y * constants::mft::InverseLayerZCoordinate()[layer1];
135  auto clsPoint2D = math_utils::Point2D<Float_t>(x_proj, y_proj);
136  r_proj = clsPoint2D.R();
137  phi_proj = clsPoint2D.Phi();
138  o2::math_utils::bringTo02PiGen(phi_proj);
139 
140  binR_proj = getRBinIndex(r_proj);
141  binPhi_proj = getPhiBinIndex(phi_proj);
142 
143  int binRS, binPhiS;
144 
145  int binwRS = mLTFseed2BinWin;
146  int binhwRS = binwRS / 2;
147 
148  int binwPhiS = mLTFseed2BinWin;
149  int binhwPhiS = binwPhiS / 2;
150 
151  for (Int_t iR = 0; iR < binwRS; ++iR) {
152  binRS = binR_proj + (iR - binhwRS);
153  if (binRS < 0) {
154  continue;
155  }
156 
157  for (Int_t iPhi = 0; iPhi < binwPhiS; ++iPhi) {
158  binPhiS = binPhi_proj + (iPhi - binhwPhiS);
159  if (binPhiS < 0) {
160  continue;
161  }
162 
163  binIndex2S = getBinIndex(binRS, binPhiS);
164  mBinsS[layer1][layer2 - 1][binIndex1].emplace_back(binIndex2S);
165  }
166  }
167 
168  int binR, binPhi;
169 
170  int binwR = mLTFinterBinWin;
171  int binhwR = binwR / 2;
172 
173  int binwPhi = mLTFinterBinWin;
174  int binhwPhi = binwPhi / 2;
175 
176  for (Int_t iR = 0; iR < binwR; ++iR) {
177  binR = binR_proj + (iR - binhwR);
178  if (binR < 0) {
179  continue;
180  }
181 
182  for (Int_t iPhi = 0; iPhi < binwPhi; ++iPhi) {
183  binPhi = binPhi_proj + (iPhi - binhwPhi);
184  if (binPhi < 0) {
185  continue;
186  }
187 
188  binIndex2 = getBinIndex(binR, binPhi);
189  mBins[layer1][layer2 - 1][binIndex1].emplace_back(binIndex2);
190  }
191  }
192 
193  } // end loop layer2
194  } // end loop PhiBinIndex
195  } // end loop RBinIndex
196  } // end loop layer1
197 }

222 template <typename T>
223 void Tracker<T>::findTracksLTF(ROframe<T>& event)
224 {
225  // find (high momentum) tracks by the Linear Track Finder (LTF) method
226 
227  MCCompLabel mcCompLabel;
228  Int_t layer1, layer2, nPointDisks;
229  Int_t binR_proj, binPhi_proj, bin;
230  Int_t binIndex, clsMinIndex, clsMaxIndex, clsMinIndexS, clsMaxIndexS;
231  Int_t extClsIndex;
232  Float_t dz = 0., dRCone = 1.;
233  Float_t dR2, dR2min, dR2cut = mLTFclsR2Cut;
234  Bool_t hasDisk[constants::mft::DisksNumber], newPoint;
235 
236  Int_t clsInLayer1, clsInLayer2, clsInLayer;
237 
238  Int_t nPoints;
239  TrackElement trackPoints[constants::mft::LayersNumber];
240 
241  Int_t step = 0;
242  layer1 = 0;
243 
244  while (true) {
245 
246  layer2 = (step == 0) ? (constants::mft::LayersNumber - 1) : (layer2 - 1);
247  step++;
248 
249  if (layer2 < layer1 + (mMinTrackPointsLTF - 1)) {
250  ++layer1;
251  if (layer1 > (constants::mft::LayersNumber - (mMinTrackPointsLTF - 1))) {
252  break;
253  }
254  step = 0;
255  continue;
256  }
257 
258  for (std::vector<Cluster>::iterator it1 = event.getClustersInLayer(layer1).begin(); it1 != event.getClustersInLayer(layer1).end(); ++it1) {
259  Cluster& cluster1 = *it1;
260  if (cluster1.getUsed()) {
261  continue;
262  }
263  clsInLayer1 = it1 - event.getClustersInLayer(layer1).begin();
264 
265  // loop over the bins in the search window
266  for (auto& binS : mBinsS[layer1][layer2 - 1][cluster1.indexTableBin]) {
267 
268  getBinClusterRange(event, layer2, binS, clsMinIndexS, clsMaxIndexS);
269 
270  for (std::vector<Cluster>::iterator it2 = (event.getClustersInLayer(layer2).begin() + clsMinIndexS); it2 != (event.getClustersInLayer(layer2).begin() + clsMaxIndexS + 1); ++it2) {
271  Cluster& cluster2 = *it2;
272  if (cluster2.getUsed()) {
273  continue;
274  }
275  clsInLayer2 = it2 - event.getClustersInLayer(layer2).begin();
276 
277  // start a Track type T
278  nPoints = 0;
279 
280  // add the first seed point
281  trackPoints[nPoints].layer = layer1;
282  trackPoints[nPoints].idInLayer = clsInLayer1;
283  nPoints++;
284 
285  // intermediate layers
286  for (Int_t layer = (layer1 + 1); layer <= (layer2 - 1); ++layer) {
287 
288  newPoint = kTRUE;
289 
290  //check if road is a cylinder or a cone
291  dz = constants::mft::LayerZCoordinate()[layer2] - constants::mft::LayerZCoordinate()[layer1];
292  dRCone = 1 + dz * constants::mft::InverseLayerZCoordinate()[layer1];
293 
294  // loop over the bins in the search window
295  dR2min = mLTFConeRadius ? dR2cut * dRCone * dRCone : dR2cut;
296  for (auto& bin : mBins[layer1][layer - 1][cluster1.indexTableBin]) {
297 
298  getBinClusterRange(event, layer, bin, clsMinIndex, clsMaxIndex);
299 
300  for (std::vector<Cluster>::iterator it = (event.getClustersInLayer(layer).begin() + clsMinIndex); it != (event.getClustersInLayer(layer).begin() + clsMaxIndex + 1); ++it) {
301  Cluster& cluster = *it;
302  if (cluster.getUsed()) {
303  continue;
304  }
305  clsInLayer = it - event.getClustersInLayer(layer).begin();
306 
307  dR2 = getDistanceToSeed(cluster1, cluster2, cluster);
308  // retain the closest point within a radius dR2cut
309  if (dR2 >= dR2min) {
310  continue;
311  }
312  dR2min = dR2;
313 
314  if (newPoint) {
315  trackPoints[nPoints].layer = layer;
316  trackPoints[nPoints].idInLayer = clsInLayer;
317  nPoints++;
318  }
319  // retain only the closest point in DistanceToSeed
320  newPoint = false;
321  } // end clusters bin intermediate layer
322  } // end intermediate layers
323  } // end binRPhi
324 
325  // add the second seed point
326  trackPoints[nPoints].layer = layer2;
327  trackPoints[nPoints].idInLayer = clsInLayer2;
328  nPoints++;
329 
330  // keep only tracks fulfilling the minimum length condition
331  if (nPoints < mMinTrackPointsLTF) {
332  continue;
333  }
334  for (Int_t i = 0; i < (constants::mft::DisksNumber); i++) {
335  hasDisk[i] = kFALSE;
336  }
337  for (Int_t point = 0; point < nPoints; ++point) {
338  auto layer = trackPoints[point].layer;
339  hasDisk[layer / 2] = kTRUE;
340  }
341  nPointDisks = 0;
342  for (Int_t disk = 0; disk < (constants::mft::DisksNumber); ++disk) {
343  if (hasDisk[disk]) {
344  ++nPointDisks;
345  }
346  }
347  if (nPointDisks < mMinTrackStationsLTF) {
348  continue;
349  }
350 
351  // add a new Track
352  event.addTrack();
353  for (Int_t point = 0; point < nPoints; ++point) {
354  auto layer = trackPoints[point].layer;
355  auto clsInLayer = trackPoints[point].idInLayer;
356  Cluster& cluster = event.getClustersInLayer(layer)[clsInLayer];
357  mcCompLabel = mUseMC ? event.getClusterLabels(layer, cluster.clusterId) : MCCompLabel();
358  extClsIndex = event.getClusterExternalIndex(layer, cluster.clusterId);
359  event.getCurrentTrack().setPoint(cluster, layer, clsInLayer, mcCompLabel, extClsIndex);
360  // mark the used clusters
361  cluster.setUsed(true);
362  }
363  } // end seed clusters bin layer2
364  } // end binRPhi
365  } // end clusters layer1
366 
367  } // end seeding
368 }

505 template <typename T>
506 void Tracker<T>::findTracksCA(ROframe<T>& event)
507 {
508  // layers: 0, 1, 2, ..., 9
509  // rules for combining first/last plane in a road:
510  // 0 with 6, 7, 8, 9
511  // 1 with 6, 7, 8, 9
512  // 2 with 8, 9
513  // 3 with 8, 9
514  Int_t layer1Min = 0, layer1Max = 3;
515  Int_t layer2Min[4] = {6, 6, 8, 8};
516  constexpr Int_t layer2Max = constants::mft::LayersNumber - 1;
517 
518  MCCompLabel mcCompLabel;
519  Int_t roadId, nPointDisks;
520  Int_t binR_proj, binPhi_proj, bin;
521  Int_t binIndex, clsMinIndex, clsMaxIndex, clsMinIndexS, clsMaxIndexS;
522  Float_t dz = 0., dRCone = 1.;
523  Float_t dR2, dR2min, dR2cut = mROADclsR2Cut;
524  Bool_t hasDisk[constants::mft::DisksNumber];
525 
526  Int_t clsInLayer1, clsInLayer2, clsInLayer;
527 
528  Int_t nPoints;
529  std::vector<TrackElement> roadPoints;
530 
531  roadId = 0;
532 
533  for (Int_t layer1 = layer1Min; layer1 <= layer1Max; ++layer1) {
534 
535  for (Int_t layer2 = layer2Max; layer2 >= layer2Min[layer1]; --layer2) {
536 
537  for (std::vector<Cluster>::iterator it1 = event.getClustersInLayer(layer1).begin(); it1 != event.getClustersInLayer(layer1).end(); ++it1) {
538  Cluster& cluster1 = *it1;
539  if (cluster1.getUsed()) {
540  continue;
541  }
542  clsInLayer1 = it1 - event.getClustersInLayer(layer1).begin();
543 
544  // loop over the bins in the search window
545  for (auto& binS : mBinsS[layer1][layer2 - 1][cluster1.indexTableBin]) {
546 
547  getBinClusterRange(event, layer2, binS, clsMinIndexS, clsMaxIndexS);
548 
549  for (std::vector<Cluster>::iterator it2 = (event.getClustersInLayer(layer2).begin() + clsMinIndexS); it2 != (event.getClustersInLayer(layer2).begin() + clsMaxIndexS + 1); ++it2) {
550  Cluster& cluster2 = *it2;
551  if (cluster2.getUsed()) {
552  continue;
553  }
554  clsInLayer2 = it2 - event.getClustersInLayer(layer2).begin();
555 
556  // start a road
557  roadPoints.clear();
558 
559  // add the first seed point
560  roadPoints.emplace_back(layer1, clsInLayer1);
561 
562  for (Int_t layer = (layer1 + 1); layer <= (layer2 - 1); ++layer) {
563 
564  //check if road is a cylinder or a cone
565  dz = constants::mft::LayerZCoordinate()[layer2] - constants::mft::LayerZCoordinate()[layer1];
566  dRCone = 1 + dz * constants::mft::InverseLayerZCoordinate()[layer1];
567 
568  dR2min = mLTFConeRadius ? dR2cut * dRCone * dRCone : dR2cut;
569 
570  // loop over the bins in the search window
571  for (auto& bin : mBins[layer1][layer - 1][cluster1.indexTableBin]) {
572 
573  getBinClusterRange(event, layer, bin, clsMinIndex, clsMaxIndex);
574 
575  for (std::vector<Cluster>::iterator it = (event.getClustersInLayer(layer).begin() + clsMinIndex); it != (event.getClustersInLayer(layer).begin() + clsMaxIndex + 1); ++it) {
576  Cluster& cluster = *it;
577  if (cluster.getUsed()) {
578  continue;
579  }
580  clsInLayer = it - event.getClustersInLayer(layer).begin();
581 
582  dR2 = getDistanceToSeed(cluster1, cluster2, cluster);
583  // add all points within a radius dR2cut
584  if (dR2 >= dR2min) {
585  continue;
586  }
587 
588  roadPoints.emplace_back(layer, clsInLayer);
589 
590  } // end clusters bin intermediate layer
591  } // end intermediate layers
592  } // end binR
593 
594  // add the second seed point
595  roadPoints.emplace_back(layer2, clsInLayer2);
596  nPoints = roadPoints.size();
597 
598  // keep only roads fulfilling the minimum length condition
599  if (nPoints < mMinTrackPointsCA) {
600  continue;
601  }
602  for (Int_t i = 0; i < (constants::mft::DisksNumber); i++) {
603  hasDisk[i] = kFALSE;
604  }
605  for (Int_t point = 0; point < nPoints; ++point) {
606  auto layer = roadPoints[point].layer;
607  hasDisk[layer / 2] = kTRUE;
608  }
609  nPointDisks = 0;

725  }

728  if (hasDisk[disk]) {

744

747  runBackwardInRoad(event);

814

816 
817  levelChange = kFALSE;
818  ++iter;
819 
820  // R = right, L = left
821  for (layerL = 0; layerL < (constants::mft::LayersNumber - 2); ++layerL) {
822 
823  for (icellL = 0; icellL < mRoad.getCellsInLayer(layerL).size(); ++icellL) {
824 
825  Cell& cellL = mRoad.getCellsInLayer(layerL)[icellL];
826 
827  layerR = cellL.getSecondLayerId();
828 
829  if (layerR == (constants::mft::LayersNumber - 1)) {
830  continue;
831  }
832 
833  for (icellR = 0; icellR < mRoad.getCellsInLayer(layerR).size(); ++icellR) {
834 
835  Cell& cellR = mRoad.getCellsInLayer(layerR)[icellR];
836 
837  if ((cellL.getLevel() == cellR.getLevel()) && getCellsConnect(cellL, cellR)) {
838  if (iter == 1) {
839  mRoad.addRightNeighbourToCell(layerL, icellL, layerR, icellR);
840  mRoad.addLeftNeighbourToCell(layerR, icellR, layerL, icellL);
841  }
842  mRoad.incrementCellLevel(layerR, icellR);
843  levelChange = kTRUE;
844 
845  } // end matching cells
846  } // end loop cellR
847  } // end loop cellL
848  } // end loop layer
849 
850  updateCellStatusInRoad();
851 
852  } // end while (levelChange)

856 template <typename T>
857 void Tracker<T>::runBackwardInRoad(ROframe<T>& event)
858 {
859  if (mMaxCellLevel == 1) {
860  return; // we have only isolated cells
861  }
862 
863  Bool_t addCellToNewTrack, hasDisk[constants::mft::DisksNumber];
864 
865  Int_t lastCellLayer, lastCellId, icell;
866  Int_t cellId, layerC, cellIdC, layerRC, cellIdRC, layerL, cellIdL;
867  Int_t nPointDisks;
868  Float_t deviationPrev, deviation;
869 
870  Int_t minLayer = 6;
871  Int_t maxLayer = 8;
872 
873  Int_t nCells;
874  TrackElement trackCells[constants::mft::LayersNumber - 1];
875 
876  for (Int_t layer = maxLayer; layer >= minLayer; --layer) {
877 
878  for (cellId = 0; cellId < mRoad.getCellsInLayer(layer).size(); ++cellId) {
879 
880  if (mRoad.isCellUsed(layer, cellId) || (mRoad.getCellLevel(layer, cellId) < (mMinTrackPointsCA - 1))) {
881  continue;
882  }
883 
884  // start a TrackCA
885  nCells = 0;
886 
887  trackCells[nCells].layer = layer;
888  trackCells[nCells].idInLayer = cellId;
889  nCells++;
890 
891  // add cells to the new track
892  addCellToNewTrack = kTRUE;
893  while (addCellToNewTrack) {
894 
895  layerRC = trackCells[nCells - 1].layer;
896  cellIdRC = trackCells[nCells - 1].idInLayer;
897 
898  const Cell& cellRC = mRoad.getCellsInLayer(layerRC)[cellIdRC];
899 
900  addCellToNewTrack = kFALSE;
901 
902  // loop over left neighbours
903  deviationPrev = o2::constants::math::TwoPI;
904 
905  for (Int_t iLN = 0; iLN < cellRC.getNLeftNeighbours(); ++iLN) {
906 
907  auto leftNeighbour = cellRC.getLeftNeighbours()[iLN];
908  layerL = leftNeighbour.first;
909  cellIdL = leftNeighbour.second;
910 
911  const Cell& cellL = mRoad.getCellsInLayer(layerL)[cellIdL];
912 
913  if (mRoad.isCellUsed(layerL, cellIdL) || (mRoad.getCellLevel(layerL, cellIdL) != (mRoad.getCellLevel(layerRC, cellIdRC) - 1))) {
914  continue;
915  }
916 
917  deviation = getCellDeviation(cellL, cellRC);
918 
919  if (deviation < deviationPrev) {
920 
921  deviationPrev = deviation;
922 
923  if (iLN > 0) {
924  // delete the last added cell
925  nCells--;
926  }
927 
928  trackCells[nCells].layer = layerL;
929  trackCells[nCells].idInLayer = cellIdL;
930  nCells++;
931 
932  addCellToNewTrack = kTRUE;
933  }
934 
935  } // end loop left neighbour
936  } // end while(addCellToNewTrack)
937 
938  // check the track length
939  for (Int_t i = 0; i < (constants::mft::DisksNumber); ++i) {
940  hasDisk[i] = kFALSE;
941  }
942 
943  layerC = trackCells[0].layer;
944  cellIdC = trackCells[0].idInLayer;
945  const Cell& cellC = mRoad.getCellsInLayer(layerC)[cellIdC];
946  hasDisk[cellC.getSecondLayerId() / 2] = kTRUE;
947  for (icell = 0; icell < nCells; ++icell) {
948  layerC = trackCells[icell].layer;
949  cellIdC = trackCells[icell].idInLayer;
950  hasDisk[layerC / 2] = kTRUE;
951  }
952 
953  nPointDisks = 0;
954  for (Int_t disk = 0; disk < (constants::mft::DisksNumber); ++disk) {
955  if (hasDisk[disk]) {
956  ++nPointDisks;
957  }
958  }
959 
960  if (nPointDisks < mMinTrackStationsCA) {
961  continue;
962  }
963 
964  // add a new Track setting isCA = true
965  event.addTrack(true);
966  for (icell = 0; icell < nCells; ++icell) {
967  layerC = trackCells[icell].layer;
968  cellIdC = trackCells[icell].idInLayer;
969  addCellToCurrentTrackCA(layerC, cellIdC, event);
970  mRoad.setCellUsed(layerC, cellIdC, kTRUE);
971  // marked the used clusters
972  const Cell& cellC = mRoad.getCellsInLayer(layerC)[cellIdC];
973  event.getClustersInLayer(cellC.getFirstLayerId())[cellC.getFirstClusterIndex()].setUsed(true);
974  event.getClustersInLayer(cellC.getSecondLayerId())[cellC.getSecondClusterIndex()].setUsed(true);
975  }
976  event.getCurrentTrack().sort();
977  } // end loop cells
978  } // end loop start layer
979 }
