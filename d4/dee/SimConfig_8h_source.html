35 // configuration struct (which can be passed around)
36 struct SimConfigData {
37  std::vector<std::string> mActiveModules; // list of active modules
38  std::vector<std::string> mReadoutDetectors; // list of readout detectors
39  std::string mMCEngine; // chosen VMC engine
40  std::string mGenerator; // chosen VMC generator
41  std::string mTrigger; // chosen VMC generator trigger
42  unsigned int mNEvents; // number of events to be simulated
43  std::string mExtKinFileName; // file name of external kinematics file (needed for ext kinematics generator)
44  std::string mEmbedIntoFileName; // filename containing the reference events to be used for the embedding
45  unsigned int mStartEvent; // index of first event to be taken
46  float mBMax; // maximum for impact parameter sampling
47  bool mIsMT; // chosen MT mode (Geant4 only)
48  std::string mOutputPrefix; // prefix to be used for output files
49  std::string mLogSeverity; // severity for FairLogger
50  std::string mLogVerbosity; // loglevel for FairLogger
51  std::string mKeyValueTokens; // a string holding arbitrary sequence of key-value tokens
52  // Foo.parameter1=x,Bar.parameter2=y,Baz.paramter3=hello
53  // (can be used to **loosely** change any configuration parameter from command-line)
54  std::string mConfigFile; // path to a JSON or INI config file (file extension is required to determine type).
55  // values within the config file will override values set in code by the param classes
56  // but will themselves be overridden by any values given in mKeyValueTokens.
57  int mPrimaryChunkSize; // defining max granularity for input primaries of a sim job
58  int mInternalChunkSize; //
59  int mStartSeed; // base for random number seeds
60  int mSimWorkers = 1; // number of parallel sim workers (when it applies)
61  bool mFilterNoHitEvents = false; // whether to filter out events not leaving any response
62  std::string mCCDBUrl; // the URL where to find CCDB
63  uint64_t mTimestamp; // timestamp in ms to anchor transport simulation to
64  TimeStampMode mTimestampMode = kNow; // telling of timestamp was given as option or defaulted to now
65  int mRunNumber = -1; // ALICE run number (if set != -1); the timestamp should be compatible
66  int mField; // L3 field setting in kGauss: +-2,+-5 and 0
67  SimFieldMode mFieldMode = kDefault; // uniform magnetic field
68  bool mAsService = false; // if simulation should be run as service/deamon (does not exit after run)
69  bool mNoGeant = false; // if Geant transport should be turned off (when one is only interested in the generated events)
70 
71  ClassDefNV(SimConfigData, 4);
72 };

159  std::string outputDir; // output directory
160  std::string keyValueTokens; // a string holding arbitrary sequence of key-value tokens (for ConfigurableParams)
161  // ** WE NEED TO BE CAREFUL: NOT EVERYTHING MAY BE RECONFIGURABLE VIA PARAMETER CHANGE **
162  // Foo.parameter1=x,Bar.parameter2=y,Baz.paramter3=hello
163  std::string configFile; // path to a JSON or INI config file (file extension is required to determine type).
164  // values within the config file will override values set in code by the param classes

166  unsigned int primaryChunkSize; // defining max granularity for input primaries of a sim job
