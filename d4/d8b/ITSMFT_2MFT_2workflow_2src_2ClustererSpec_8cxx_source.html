44 void ClustererDPL::init(InitContext& ic)
45 {
46 
47  mClusterer = std::make_unique<o2::itsmft::Clusterer>();
48  mClusterer->setNChips(o2::itsmft::ChipMappingMFT::getNChips());
49  mUseClusterDictionary = !ic.options().get<bool>("ignore-cluster-dictionary");
50 
51  LOG(info) << "MFT ClustererDPL::init total number of sensors " << o2::itsmft::ChipMappingMFT::getNChips() << "\n";
52 
53  //mClusterer->setMaskOverflowPixels(false);
54 
55  auto filenameGRP = ic.options().get<std::string>("grp-file");
56  const auto grp = o2::parameters::GRPObject::loadFrom(filenameGRP.c_str());
57 
58  if (grp) {
59  mClusterer->setContinuousReadOut(grp->isDetContinuousReadOut("MFT"));
60  } else {
61  LOG(error) << "Cannot retrieve GRP from the " << filenameGRP.c_str() << " file !";
62  mState = 0;
63  return;
64  }
65 
66  mPatterns = !ic.options().get<bool>("no-patterns");
67  mNThreads = std::max(1, ic.options().get<int>("nthreads"));
68 
69  mState = 1;
70 }

72 void ClustererDPL::run(ProcessingContext& pc)
73 {
74  updateTimeDependentParams(pc);
75  auto digits = pc.inputs().get<gsl::span<o2::itsmft::Digit>>("digits");
76  auto rofs = pc.inputs().get<gsl::span<o2::itsmft::ROFRecord>>("ROframes");
77 
78  gsl::span<const o2::itsmft::MC2ROFRecord> mc2rofs;
79  gsl::span<const char> labelbuffer;
80  if (mUseMC) {
81  labelbuffer = pc.inputs().get<gsl::span<char>>("labels");
82  mc2rofs = pc.inputs().get<gsl::span<o2::itsmft::MC2ROFRecord>>("MC2ROframes");
83  }
84  const o2::dataformats::ConstMCTruthContainerView<o2::MCCompLabel> labels(labelbuffer);
85 
86  LOG(debug) << "MFTClusterer pulled " << digits.size() << " digits, in "
87  << rofs.size() << " RO frames";
88 
89  o2::itsmft::DigitPixelReader reader;
90  reader.setDigits(digits);
91  reader.setROFRecords(rofs);
92  if (mUseMC) {
93  reader.setMC2ROFRecords(mc2rofs);
94  reader.setDigitsMCTruth(labels.getIndexedSize() > 0 ? &labels : nullptr);
95  }
96  reader.init();
97  auto orig = o2::header::gDataOriginMFT;
98  std::vector<o2::itsmft::CompClusterExt> clusCompVec;
99  std::vector<o2::itsmft::ROFRecord> clusROFVec;
100  std::vector<unsigned char> clusPattVec;
101 
102  std::unique_ptr<o2::dataformats::MCTruthContainer<o2::MCCompLabel>> clusterLabels;
103  if (mUseMC) {
104  clusterLabels = std::make_unique<o2::dataformats::MCTruthContainer<o2::MCCompLabel>>();
105  }
106  mClusterer->process(mNThreads, reader, &clusCompVec, mPatterns ? &clusPattVec : nullptr, &clusROFVec, clusterLabels.get());
107  pc.outputs().snapshot(Output{orig, "COMPCLUSTERS", 0, Lifetime::Timeframe}, clusCompVec);
108  pc.outputs().snapshot(Output{orig, "CLUSTERSROF", 0, Lifetime::Timeframe}, clusROFVec);
109  pc.outputs().snapshot(Output{orig, "PATTERNS", 0, Lifetime::Timeframe}, clusPattVec);
110 
111  if (mUseMC) {
112  pc.outputs().snapshot(Output{orig, "CLUSTERSMCTR", 0, Lifetime::Timeframe}, *clusterLabels.get()); // at the moment requires snapshot
113  std::vector<o2::itsmft::MC2ROFRecord> clusterMC2ROframes(mc2rofs.size());
114  for (int i = mc2rofs.size(); i--;) {
115  clusterMC2ROframes[i] = mc2rofs[i]; // Simply, replicate it from digits ?
116  }
117  pc.outputs().snapshot(Output{orig, "CLUSTERSMC2ROF", 0, Lifetime::Timeframe}, clusterMC2ROframes);
118  }
119 
120  // TODO: in principle, after masking "overflow" pixels the MC2ROFRecord maxROF supposed to change, nominally to minROF
121  // -> consider recalculationg maxROF
122  LOG(debug) << "MFTClusterer pushed " << clusCompVec.size() << " compressed clusters, in " << clusROFVec.size() << " RO frames";
123 }

126 void ClustererDPL::updateTimeDependentParams(ProcessingContext& pc)
127 {
128  static bool initOnceDone = false;
129  if (!initOnceDone) { // this params need to be queried only once
130  initOnceDone = true;
131  pc.inputs().get<TopologyDictionary*>("cldict"); // just to trigger the finaliseCCDB
132  pc.inputs().get<o2::itsmft::DPLAlpideParam<o2::detectors::DetID::MFT>*>("alppar");
133  pc.inputs().get<o2::itsmft::ClustererParam<o2::detectors::DetID::MFT>*>("cluspar");
134 
135  // settings for the fired pixel overflow masking
136  const auto& alpParams = o2::itsmft::DPLAlpideParam<o2::detectors::DetID::MFT>::Instance();
137  const auto& clParams = o2::itsmft::ClustererParam<o2::detectors::DetID::MFT>::Instance();
138  auto nbc = clParams.maxBCDiffToMaskBias;
139  nbc += mClusterer->isContinuousReadOut() ? alpParams.roFrameLengthInBC : (alpParams.roFrameLengthTrig / o2::constants::lhc::LHCBunchSpacingNS);
140  mClusterer->setMaxBCSeparationToMask(nbc);
141  mClusterer->setMaxRowColDiffToMask(clParams.maxRowColDiffToMask);
142  mClusterer->print();
143  }
144  // we may have other params which need to be queried regularly
145 }

148 void ClustererDPL::finaliseCCDB(ConcreteDataMatcher& matcher, void* obj)
149 {
150  if (matcher == ConcreteDataMatcher("MFT", "CLUSDICT", 0)) {
151  LOG(info) << "cluster dictionary updated" << (!mUseClusterDictionary ? " but its using is disabled" : "");
152  if (mUseClusterDictionary) {
153  mClusterer->setDictionary((const TopologyDictionary*)obj);
154  }
155  return;
156  }
157  // Note: strictly speaking, for Configurable params we don't need finaliseCCDB check, the singletons are updated at the CCDB fetcher level
158  if (matcher == ConcreteDataMatcher("MFT", "ALPIDEPARAM", 0)) {
159  LOG(info) << "Alpide param updated";
160  const auto& par = o2::itsmft::DPLAlpideParam<o2::detectors::DetID::MFT>::Instance();
161  par.printKeyValues();
162  return;
163  }
164  if (matcher == ConcreteDataMatcher("MFT", "CLUSPARAM", 0)) {
165  LOG(info) << "Cluster param updated";
166  const auto& par = o2::itsmft::ClustererParam<o2::detectors::DetID::MFT>::Instance();
167  par.printKeyValues();
168  return;
169  }

172 DataProcessorSpec getClustererSpec(bool useMC)
173 {
174  std::vector<InputSpec> inputs;
175  inputs.emplace_back("digits", "MFT", "DIGITS", 0, Lifetime::Timeframe);
176  inputs.emplace_back("ROframes", "MFT", "DIGITSROF", 0, Lifetime::Timeframe);
177  inputs.emplace_back("cldict", "MFT", "CLUSDICT", 0, Lifetime::Condition, ccdbParamSpec("MFT/Calib/ClusterDictionary"));
178  inputs.emplace_back("cluspar", "MFT", "CLUSPARAM", 0, Lifetime::Condition, ccdbParamSpec("MFT/Config/ClustererParam"));
179  inputs.emplace_back("alppar", "MFT", "ALPIDEPARAM", 0, Lifetime::Condition, ccdbParamSpec("MFT/Config/AlpideParam"));
180 
181  std::vector<OutputSpec> outputs;
182  outputs.emplace_back("MFT", "COMPCLUSTERS", 0, Lifetime::Timeframe);
183  outputs.emplace_back("MFT", "PATTERNS", 0, Lifetime::Timeframe);
184  outputs.emplace_back("MFT", "CLUSTERSROF", 0, Lifetime::Timeframe);
185 
186  if (useMC) {
187  inputs.emplace_back("labels", "MFT", "DIGITSMCTR", 0, Lifetime::Timeframe);
188  inputs.emplace_back("MC2ROframes", "MFT", "DIGITSMC2ROF", 0, Lifetime::Timeframe);
189  outputs.emplace_back("MFT", "CLUSTERSMCTR", 0, Lifetime::Timeframe);
190  outputs.emplace_back("MFT", "CLUSTERSMC2ROF", 0, Lifetime::Timeframe);
191  }
192 
193  return DataProcessorSpec{
194  "mft-clusterer",
195  inputs,
196  outputs,
197  AlgorithmSpec{adaptFromTask<ClustererDPL>(useMC)},
198  Options{
199  {"grp-file", VariantType::String, "o2sim_grp.root", {"Name of the grp file"}},
200  {"no-patterns", o2::framework::VariantType::Bool, false, {"Do not save rare cluster patterns"}},
201  {"ignore-cluster-dictionary", VariantType::Bool, false, {"do not use cluster dictionary, always store explicit patterns"}},
202  {"nthreads", VariantType::Int, 1, {"Number of clustering threads"}}}};
203 }
