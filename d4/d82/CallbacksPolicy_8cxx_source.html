25 CallbacksPolicy epnProcessReporting()
26 {
27  return {
28  .matcher = [](DeviceSpec const&, ConfigContext const& context) -> bool {
30  static bool report = getenv("DDS_SESSION_ID") != nullptr || getenv("DPL_REPORT_PROCESSING") != nullptr;
31  return report;
32  },
33  .policy = [](CallbackService& callbacks, InitContext& context) -> void {
34  callbacks.set(CallbackService::Id::PreProcessing, [](ServiceRegistry& registry, int op) {
35  auto& info = registry.get<TimingInfo>();
36  if ((int)info.firstTFOrbit != -1) {
37  char const* what = (info.timeslice > 1652945069870351) ? "timer" : "timeslice";
38  LOGP(info, "Processing {}:{}, tfCounter:{}, firstTFOrbit:{}, runNumber:{}, creation:{}, action:{}",
39  what, info.timeslice, info.tfCounter, info.firstTFOrbit, info.runNumber, info.creation, op);
40  }
41  });
42  callbacks.set(CallbackService::Id::PostProcessing, [](ServiceRegistry& registry, int op) {
43  auto& info = registry.get<TimingInfo>();
44  if ((int)info.firstTFOrbit != -1) {
45  char const* what = (info.timeslice > 1652945069870351) ? "timer" : "timeslice";
46  LOGP(info, "Done processing {}:{}, tfCounter:{}, firstTFOrbit:{}, runNumber:{}, creation:{}, action:{}",
47  what, info.timeslice, info.tfCounter, info.firstTFOrbit, info.runNumber, info.creation, op);
48  }
49  });
50  }};
51 }
