31 std::vector<std::pair<VisualisationEvent, EVisualisationGroup>> DataSourceOnline::getVisualisationList(int no, float minTime, float maxTime, float range)
32 {
33  std::vector<std::pair<VisualisationEvent, EVisualisationGroup>> res;
34  if (no < getEventCount()) {
35  assert(no >= 0);
36 
37  mFileWatcher.setCurrentItem(no);
38  VisualisationEvent vEvent = this->mDataReader->getEvent(mFileWatcher.currentFilePath());
39 
40  this->setRunNumber(vEvent.getRunNumber());
41  this->setFirstTForbit(vEvent.getFirstTForbit());
42  this->setCollisionTime(vEvent.getCollisionTime());
43  this->setTrackMask(vEvent.getTrkMask());
44  this->setClusterMask(vEvent.getClMask());
45 
46  double period = vEvent.getMaxTimeOfTracks() - vEvent.getMinTimeOfTracks();
47  if (period > 0) {
48  this->mTimeFrameMinTrackTime = minTime * period / range + vEvent.getMinTimeOfTracks();
49  this->mTimeFrameMaxTrackTime = maxTime * period / range + vEvent.getMinTimeOfTracks();
50  } else {
51  this->mTimeFrameMinTrackTime = vEvent.getMinTimeOfTracks();
52  this->mTimeFrameMaxTrackTime = vEvent.getMaxTimeOfTracks();
53  }
54 
55  for (auto filter = EVisualisationGroup::ITS;
56  filter != EVisualisationGroup::NvisualisationGroups;
57  filter = static_cast<EVisualisationGroup>(static_cast<int>(filter) + 1)) {
58  auto filtered = VisualisationEvent(vEvent, filter, this->mTimeFrameMinTrackTime, this->mTimeFrameMaxTrackTime);
59  res.push_back(std::make_pair(filtered, filter)); // we can switch on/off data
60  }
61  }
62  return res;
63 }
