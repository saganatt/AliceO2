117 template <typename DataInput, typename DataOutput, typename HistContainer>
118 void EMCALChannelCalibrator<DataInput, DataOutput, HistContainer>::finalizeSlot(o2::calibration::TimeSlot<DataInput>& slot)
119 {
120 
121  // Extract results for the single slot
122  DataInput* c = slot.getContainer();
123  LOG(info) << "Finalize slot " << slot.getTFStart() << " <= TF <= " << slot.getTFEnd();
124 
125  std::map<std::string, std::string> md;
126  if constexpr (std::is_same<DataInput, o2::emcal::EMCALChannelData>::value) {
127  auto bcm = mCalibrator->calibrateBadChannels(c->getHisto());
128  // for the CCDB entry
129  auto clName = o2::utils::MemFileHelper::getClassName(bcm);
130  auto flName = o2::ccdb::CcdbApi::generateFileName(clName);
131  mInfoVector.emplace_back(CalibDB::getCDBPathBadChannelMap(), clName, flName, md, slot.getStartTimeMS(), o2::ccdb::CcdbObjectInfo::INFINITE_TIMESTAMP);
132  mCalibObjectVector.push_back(bcm);
133  } else if constexpr (std::is_same<DataInput, o2::emcal::EMCALTimeCalibData>::value) {
134  auto tcd = mCalibrator->calibrateTime(c->getHisto());
135 
136  // for the CCDB entry
137  auto clName = o2::utils::MemFileHelper::getClassName(slot);
138  auto flName = o2::ccdb::CcdbApi::generateFileName(clName);
139 
140  //prepareCCDBobjectInfo
141  mInfoVector.emplace_back(CalibDB::getCDBPathTimeCalibrationParams(), clName, flName, md, slot.getStartTimeMS(), o2::ccdb::CcdbObjectInfo::INFINITE_TIMESTAMP);
142  mCalibObjectVector.push_back(tcd);
143 
144  if ((EMCALCalibParams::Instance().localRootFilePath).find(".root") != std::string::npos) {
145  TFile fLocalStorage((EMCALCalibParams::Instance().localRootFilePath).c_str(), "update");
146  fLocalStorage.cd();
147  TH1F* histTCparams = (TH1F*)tcd.getHistogramRepresentation(false);
148  std::string nameTCHist = "TCParams_" + std::to_string(slot.getStartTimeMS());
149  histTCparams->Write(nameTCHist.c_str(), TObject::kOverwrite);
150 
151  TH2F hCalibHist = o2::utils::TH2FFromBoost(c->getHisto());
152  std::string nameTCInputHist = "TimeVsCellID_" + std::to_string(slot.getStartTimeMS());
153  hCalibHist.Write(nameTCInputHist.c_str(), TObject::kOverwrite);
154  fLocalStorage.Close();
155  }
156  }
157 }

159 template <typename DataInput, typename DataOutput, typename HistContainer>
160 o2::calibration::TimeSlot<DataInput>& EMCALChannelCalibrator<DataInput, DataOutput, HistContainer>::emplaceNewSlot(bool front, TFType tstart, TFType tend)
161 {
162  auto& cont = o2::calibration::TimeSlotCalibration<o2::emcal::Cell, DataInput>::getSlots();
163  auto& slot = front ? cont.emplace_front(tstart, tend) : cont.emplace_back(tstart, tend);
164  HistContainer histcont; // initialize struct with (default) ranges for time or channel calibration.
165  slot.setContainer(std::make_unique<DataInput>(histcont));
166  return slot;
167 }
