61 void RawToDigitConverterSpec::run(framework::ProcessingContext& ctx)
62 {
63  // check timers if we need mute/unmute error reporting
64  auto now = std::chrono::system_clock::now();
65  if (mIsMuteDecoderErrors) { // check if 10-minutes muting period passed
66  if (((now - mTimeWhenMuted) / std::chrono::minutes(1)) >= 10) {
67  mIsMuteDecoderErrors = false; // unmute
68  if (mDecoderErrorsCounterWhenMuted) {
69  LOG(error) << "RawToDigitConverterSpec::run() : " << mDecoderErrorsCounterWhenMuted << " errors happened while it was muted ((";
70  }
71  mDecoderErrorsCounterWhenMuted = 0;
72  }
73  }
74  if (((now - mStartTime) / std::chrono::minutes(1)) > mMinutesPassed) {
75  mMinutesPassed = (now - mStartTime) / std::chrono::minutes(1);
76  LOG(debug) << "minutes passed: " << mMinutesPassed;
77  mDecoderErrorsPerMinute = 0;
78  }
79 
80  // Cache digits from bunch crossings as the component reads timeframes from many links consecutively
81  std::map<o2::InteractionRecord, std::shared_ptr<std::vector<o2::cpv::Digit>>> digitBuffer; // Internal digit buffer
82  mOutputHWErrors.clear();
83 
84  // if we see requested data type input with 0xDEADBEEF subspec and 0 payload this means that the "delayed message"
85  // mechanism created it in absence of real data from upstream. Processor should send empty output to not block the workflow
86  static size_t contDeadBeef = 0; // number of times 0xDEADBEEF was seen continuously
87  std::vector<o2::framework::InputSpec> dummy{o2::framework::InputSpec{"dummy", o2::framework::ConcreteDataMatcher{"CPV", o2::header::gDataDescriptionRawData, 0xDEADBEEF}}};
88  for (const auto& ref : framework::InputRecordWalker(ctx.inputs(), dummy)) {
89  const auto dh = o2::framework::DataRefUtils::getHeader<o2::header::DataHeader*>(ref);
90  auto payloadSize = o2::framework::DataRefUtils::getPayloadSize(ref);
91  if (payloadSize == 0) { // send empty output
92  auto maxWarn = o2::conf::VerbosityConfig::Instance().maxWarnDeadBeef;
93  if (++contDeadBeef <= maxWarn) {
94  LOGP(alarm, "Found input [{}/{}/{:#x}] TF#{} 1st_orbit:{} Payload {} : assuming no payload for all links in this TF{}",
95  dh->dataOrigin.str, dh->dataDescription.str, dh->subSpecification, dh->tfCounter, dh->firstTForbit, payloadSize,
96  contDeadBeef == maxWarn ? fmt::format(". {} such inputs in row received, stopping reporting", contDeadBeef) : "");
97  }
98  mOutputDigits.clear();
99  ctx.outputs().snapshot(o2::framework::Output{"CPV", "DIGITS", 0, o2::framework::Lifetime::Timeframe}, mOutputDigits);
100  mOutputTriggerRecords.clear();
101  ctx.outputs().snapshot(o2::framework::Output{"CPV", "DIGITTRIGREC", 0, o2::framework::Lifetime::Timeframe}, mOutputTriggerRecords);
102  mOutputHWErrors.clear();
103  ctx.outputs().snapshot(o2::framework::Output{"CPV", "RAWHWERRORS", 0, o2::framework::Lifetime::Timeframe}, mOutputHWErrors);
104  return; // empty TF, nothing to process
105  }
106  }
107  contDeadBeef = 0; // if good data, reset the counter
108 
109  const o2::cpv::Pedestals* pedestals = nullptr;
110  const o2::cpv::BadChannelMap* badMap = nullptr;
111  const o2::cpv::CalibParams* gains = nullptr;
112  std::decay_t<decltype(ctx.inputs().get<o2::cpv::Pedestals*>("peds"))> pedPtr{};
113  std::decay_t<decltype(ctx.inputs().get<o2::cpv::BadChannelMap*>("badmap"))> badMapPtr{};
114  std::decay_t<decltype(ctx.inputs().get<o2::cpv::CalibParams*>("gains"))> gainsPtr{};
115 
116  if (!mIsPedestalData) {
117  pedPtr = ctx.inputs().get<o2::cpv::Pedestals*>("peds");
118  pedestals = pedPtr.get();
119  }
120 
121  if (mIsUsingBadMap) {
122  badMapPtr = ctx.inputs().get<o2::cpv::BadChannelMap*>("badmap");
123  badMap = badMapPtr.get();
124  }
125 
126  if (mIsUsingGainCalibration) {
127  gainsPtr = ctx.inputs().get<o2::cpv::CalibParams*>("gains");
128  gains = gainsPtr.get();
129  }
130 
131  bool skipTF = false; // skip TF due to fatal error?
132 
133  std::vector<o2::framework::InputSpec> rawFilter{
134  {"RAWDATA", o2::framework::ConcreteDataTypeMatcher{"CPV", "RAWDATA"}, o2::framework::Lifetime::Timeframe},
135  };
136  for (const auto& rawData : framework::InputRecordWalker(ctx.inputs(), rawFilter)) {
137  o2::cpv::RawReaderMemory rawreader(o2::framework::DataRefUtils::as<const char>(rawData));
138  // loop over all the DMA pages
139  while (rawreader.hasNext()) {
140  try {
141  rawreader.next();
142  } catch (RawErrorType_t e) {
143  if (!mIsMuteDecoderErrors) {
144  LOG(error) << "Raw decoding error " << (int)e;
145  }
146  // add error list
147  // RawErrorType_t is defined in O2/Detectors/CPV/reconstruction/include/CPVReconstruction/RawReaderMemory.h
148  // RawDecoderError(short c, short d, short g, short p, RawErrorType_t e)
149  mOutputHWErrors.emplace_back(-1, 0, 0, 0, e); // Put general errors to non-existing ccId -1
150  // if problem in header, abandon this page
151  if (e == RawErrorType_t::kRDH_DECODING) { // fatal error -> skip whole TF
152  LOG(error) << "RDH decoding error. Skipping this TF";
153  skipTF = true;
154  break;
155  }
156  if (e == RawErrorType_t::kPAGE_NOTFOUND || // nothing left to read -> skip to next HBF
157  e == RawErrorType_t::kNOT_CPV_RDH || // not cpv rdh -> skip to next HBF
158  e == RawErrorType_t::kOFFSET_TO_NEXT_IS_0) { // offset to next package is 0 -> do not know how to read next -> skip to next HBF
159  break;
160  }
161  // if problem in payload, try to continue
162  continue;
163  }
164  auto& rdh = rawreader.getRawHeader();
165  auto triggerOrbit = o2::raw::RDHUtils::getTriggerOrbit(rdh);
166  auto mod = o2::raw::RDHUtils::getLinkID(rdh) + 2; // link=0,1,2 -> mod=2,3,4
167  // for now all modules are written to one LinkID
168  if (mod > o2::cpv::Geometry::kNMod || mod < 2) { // only 3 correct modules:2,3,4
169  if (!mIsMuteDecoderErrors) {
170  LOG(error) << "RDH linkId corresponds to module " << mod << " which does not exist";
171  }
172  mOutputHWErrors.emplace_back(-1, mod, 0, 0, kRDH_INVALID); // Add non-existing modules to non-existing ccId -1 and dilogic = mod
173  continue;
174  }
175  o2::cpv::RawDecoder decoder(rawreader);
176  if (mIsMuteDecoderErrors) {
177  decoder.muteErrors();
178  }
179  RawErrorType_t err = decoder.decode();
180  int decoderErrors = 0;
181  for (auto errs : decoder.getErrors()) {
182  if (errs.ccId == -1) { // error related to wrong data format
183  decoderErrors++;
184  }
185  }
186  mDecoderErrorsPerMinute += decoderErrors;
187  // LOG(debug) << "RawDecoder found " << decoderErrors << " raw format errors";
188  // LOG(debug) << "Now I have " << mDecoderErrorsPerMinute << " errors for current minute";
189  if (mIsMuteDecoderErrors) {
190  mDecoderErrorsCounterWhenMuted += decoder.getErrors().size();
191  } else {
192  if (mDecoderErrorsPerMinute > 10) { // mute error reporting for 10 minutes
193  LOG(warning) << "> 10 raw decoder error messages per minute, muting it for 10 minutes";
194  mIsMuteDecoderErrors = true;
195  mTimeWhenMuted = std::chrono::system_clock::now();
196  }
197  }
198 
199  if (!(err == kOK || err == kOK_NO_PAYLOAD)) {
200  // TODO handle severe errors
201  // TODO: probably careful conversion of decoder errors to Fitter errors?
202  mOutputHWErrors.emplace_back(-1, mod, 0, 0, err); // assign general RDH errors to non-existing ccId -1 and dilogic = mod
203  }
204 
205  std::shared_ptr<std::vector<o2::cpv::Digit>> currentDigitContainer;
206  auto digilets = decoder.getDigits();
207  if (digilets.empty()) { // no digits -> continue to next pages
208  continue;
209  }
210  o2::InteractionRecord currentIR(0, triggerOrbit); //(bc, orbit)
211  // Loop over all the BCs
212  for (auto itBCRecords : decoder.getBCRecords()) {
213  currentIR.bc = itBCRecords.bc;
214  for (unsigned int iDig = itBCRecords.firstDigit; iDig <= itBCRecords.lastDigit; iDig++) {
215  auto adch = digilets[iDig];
216  auto found = digitBuffer.find(currentIR);
217  if (found == digitBuffer.end()) {
218  currentDigitContainer = std::make_shared<std::vector<o2::cpv::Digit>>();
219  digitBuffer[currentIR] = currentDigitContainer;
220  } else {
221  currentDigitContainer = found->second;
222  }
223 
224  AddressCharge ac = {adch};
225  unsigned short absId = ac.Address;
226  // if we deal with non-pedestal data?
227  if (!mIsPedestalData) { // not a pedestal data
228  // test bad map
229  if (mIsUsingBadMap) {
230  if (!badMap->isChannelGood(absId)) {
231  continue; // skip bad channel
232  }
233  }
234  float amp = 0;
235  if (mIsUsingGainCalibration) { // calibrate amplitude
236  amp = gains->getGain(absId) * (ac.Charge - pedestals->getPedestal(absId));
237  } else { // no gain calibration needed
238  amp = ac.Charge - pedestals->getPedestal(absId);
239  }
240  if (amp > 0) { // emplace new digit
241  currentDigitContainer->emplace_back(absId, amp, -1);
242  }
243  } else { // pedestal data, no calibration needed.
244  currentDigitContainer->emplace_back(absId, (float)ac.Charge, -1);
245  }
246  }
247  }
248  // Check and send list of hwErrors
249  for (auto& er : decoder.getErrors()) {
250  mOutputHWErrors.push_back(er);
251  }
252  }
253  // if corrupted raw data is present in this TF -> skip it
254  if (skipTF) {
255  // Send no digits
256  mOutputDigits.clear();
257  ctx.outputs().snapshot(o2::framework::Output{"CPV", "DIGITS", 0, o2::framework::Lifetime::Timeframe}, mOutputDigits);
258  mOutputTriggerRecords.clear();
259  ctx.outputs().snapshot(o2::framework::Output{"CPV", "DIGITTRIGREC", 0, o2::framework::Lifetime::Timeframe}, mOutputTriggerRecords);
260  // Send errors
261  ctx.outputs().snapshot(o2::framework::Output{"CPV", "RAWHWERRORS", 0, o2::framework::Lifetime::Timeframe}, mOutputHWErrors);
262  return;
263  }
264  }
265 
266  // Loop over BCs, sort digits with increasing digit ID and write to output containers
267  mOutputDigits.clear();
268  mOutputTriggerRecords.clear();
269  for (auto [bc, digits] : digitBuffer) {
270  int prevDigitSize = mOutputDigits.size();
271  if (digits->size()) {
272  // Sort digits according to digit ID
273  std::sort(digits->begin(), digits->end(), [](o2::cpv::Digit& lhs, o2::cpv::Digit& rhs) { return lhs.getAbsId() < rhs.getAbsId(); });
274 
275  for (auto digit : *digits) {
276  mOutputDigits.push_back(digit);
277  }
278  }
279 
280  mOutputTriggerRecords.emplace_back(bc, prevDigitSize, mOutputDigits.size() - prevDigitSize);
281  }
282  digitBuffer.clear();
283 
284  LOG(info) << "[CPVRawToDigitConverter - run] Sending " << mOutputDigits.size() << " digits in " << mOutputTriggerRecords.size() << "trigger records.";
285  ctx.outputs().snapshot(o2::framework::Output{"CPV", "DIGITS", 0, o2::framework::Lifetime::Timeframe}, mOutputDigits);
286  ctx.outputs().snapshot(o2::framework::Output{"CPV", "DIGITTRIGREC", 0, o2::framework::Lifetime::Timeframe}, mOutputTriggerRecords);
287  ctx.outputs().snapshot(o2::framework::Output{"CPV", "RAWHWERRORS", 0, o2::framework::Lifetime::Timeframe}, mOutputHWErrors);
288 }

290 o2::framework::DataProcessorSpec o2::cpv::reco_workflow::getRawToDigitConverterSpec(bool askDISTSTF, bool isPedestal, bool useBadChannelMap, bool useGainCalibration)
291 {
292  std::vector<o2::framework::InputSpec> inputs;
293  inputs.emplace_back("RAWDATA", o2::framework::ConcreteDataTypeMatcher{"CPV", "RAWDATA"}, o2::framework::Lifetime::Optional);
294  // receive at least 1 guaranteed input (which will allow to acknowledge the TF)
295  if (askDISTSTF) {
296  inputs.emplace_back("STFDist", "FLP", "DISTSUBTIMEFRAME", 0, o2::framework::Lifetime::Timeframe);
297  }
298  if (!isPedestal) {
299  inputs.emplace_back("peds", "CPV", "CPV_Pedestals", 0, o2::framework::Lifetime::Condition, o2::framework::ccdbParamSpec("CPV/Calib/Pedestals"));
300  if (useBadChannelMap) {
301  inputs.emplace_back("badmap", "CPV", "CPV_BadMap", 0, o2::framework::Lifetime::Condition, o2::framework::ccdbParamSpec("CPV/Calib/BadChannelMap"));
302  }
303  if (useGainCalibration) {
304  inputs.emplace_back("gains", "CPV", "CPV_Gains", 0, o2::framework::Lifetime::Condition, o2::framework::ccdbParamSpec("CPV/Calib/Gains"));
305  }
306  }
307 
308  std::vector<o2::framework::OutputSpec> outputs;
309  outputs.emplace_back("CPV", "DIGITS", 0, o2::framework::Lifetime::Timeframe);
310  outputs.emplace_back("CPV", "DIGITTRIGREC", 0, o2::framework::Lifetime::Timeframe);
311  outputs.emplace_back("CPV", "RAWHWERRORS", 0, o2::framework::Lifetime::Timeframe);
312  // note that for cpv we always have stream #0 (i.e. CPV/DIGITS/0)
313 
314  return o2::framework::DataProcessorSpec{"CPVRawToDigitConverterSpec",
315  inputs, // o2::framework::select("A:CPV/RAWDATA"),
316  outputs,
317  o2::framework::adaptFromTask<o2::cpv::reco_workflow::RawToDigitConverterSpec>(isPedestal, useBadChannelMap, useGainCalibration),
318  o2::framework::Options{}};
319 }
