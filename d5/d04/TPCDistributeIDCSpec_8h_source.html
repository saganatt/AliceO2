46  TPCDistributeIDCSpec(const std::vector<uint32_t>& crus, const unsigned int timeframes, const unsigned int outlanes, const bool loadFromFile, const int firstTF)
47  : mCRUs{crus}, mTimeFrames{timeframes}, mOutLanes{outlanes}, mLoadFromFile{loadFromFile}, mProcessedCRU{{std::vector<unsigned int>(timeframes), std::vector<unsigned int>(timeframes)}}, mDataSent{std::vector<bool>(timeframes), std::vector<bool>(timeframes)}, mTFStart{{firstTF, firstTF + timeframes}}, mTFEnd{{firstTF + timeframes - 1, mTFStart[1] + timeframes - 1}}
48  {
49  // sort vector for binary_search
50  std::sort(mCRUs.begin(), mCRUs.end());
51 
52  for (auto& idcbuffer : mIDCs) {
53  for (auto& idc : idcbuffer) {
54  idc.resize(mTimeFrames);
55  }
56  }
57 
58  for (auto& processedCRUbuffer : mProcessedCRUs) {
59  processedCRUbuffer.resize(mTimeFrames);
60  for (auto& crusMap : processedCRUbuffer) {
61  crusMap.reserve(mCRUs.size());
62  for (const auto cruID : mCRUs) {
63  crusMap.emplace(cruID, false);
64  }
65  }
66  }

69  void init(o2::framework::InitContext& ic) final
70  {
71  if (mLoadFromFile) {
72  TFile fInp("IDCGroup.root", "READ");
73  for (TObject* keyAsObj : *fInp.GetListOfKeys()) {
74  const auto key = dynamic_cast<TKey*>(keyAsObj);
75  LOGP(info, "Key name: {} Type: {}", key->GetName(), key->GetClassName());
76 
77  if (std::strcmp(o2::tpc::IDCAverageGroup<o2::tpc::IDCAverageGroupCRU>::Class()->GetName(), key->GetClassName()) != 0) {
78  LOGP(info, "skipping object. wrong class.");
79  continue;
80  }
81  IDCAverageGroup<IDCAverageGroupCRU>* idcavg = (IDCAverageGroup<IDCAverageGroupCRU>*)fInp.Get(key->GetName());
82  unsigned int cru = idcavg->getSector() * Mapper::NREGIONS + idcavg->getRegion();
83  const std::vector<float>& idcData = idcavg->getIDCGroup().getData();
84  for (auto& idcbuffer : mIDCs) {
85  const auto nIDCS = idcData.size();
86  // TODO use memcyp etc here
87  for (auto& relTF : idcbuffer[cru]) {
88  relTF.reserve(nIDCS);
89  for (int i = 0; i < nIDCS; ++i) {
90  relTF.emplace_back(idcData[i]);
91  }
92  }
93  }
94  delete idcavg;
95  }
96  }

99  void run(o2::framework::ProcessingContext& pc) final
100  {
101  // check which buffer to use for current incoming data
102  const auto tf = processing_helpers::getCurrentTF(pc);
103 
104  // automatically detect firstTF in case firstTF was not specified
105  if (mTFStart.front() <= -1) {
106  const auto firstTF = tf;
107  const int offsetTF = std::abs(mTFStart.front() + 1);
108  mTFStart = {firstTF + offsetTF, firstTF + offsetTF + mTimeFrames};
109  mTFEnd = {mTFStart[1] - 1, mTFStart[1] - 1 + mTimeFrames};
110  LOGP(info, "Setting {} as first TF", mTFStart[0]);
111  LOGP(info, "Using offset of {} TFs for setting the first TF", offsetTF);
112  }
113 
114  const bool currentBuffer = (tf > mTFEnd[mBuffer]) ? !mBuffer : mBuffer;
115  if (mTFStart[currentBuffer] > tf) {
116  LOGP(info, "all CRUs for current TF {} already received. Skipping this TF", tf);
117  return;
118  }
119 
120  const unsigned int currentOutLane = (tf > mTFEnd[mBuffer]) ? (mCurrentOutLane + 1) % mOutLanes : mCurrentOutLane;
121  const auto relTF = tf - mTFStart[currentBuffer];
122  LOGP(info, "current TF: {} relative TF: {} current buffer: {} current output lane: {} mTFStart: {}", tf, relTF, currentBuffer, currentOutLane, mTFStart[currentBuffer]);
123 
124  if (relTF >= mProcessedCRU[currentBuffer].size()) {
125  LOGP(fatal, "Skipping tf {}: relative tf {} is larger than size of buffer: {}", tf, relTF, mProcessedCRU[currentBuffer].size());
126  return;
127  }
128 
129  if (!mLoadFromFile) {
130  for (auto& ref : InputRecordWalker(pc.inputs(), mFilter)) {
131  auto const* tpcCRUHeader = o2::framework::DataRefUtils::getHeader<o2::header::DataHeader*>(ref);
132  const int cru = tpcCRUHeader->subSpecification >> 7;
133 
134  // check if cru is specified in input cru list
135  if (!(std::binary_search(mCRUs.begin(), mCRUs.end(), cru))) {
136  LOGP(debug, "Received data from CRU: {} which was not specified as input. Skipping", cru);
137  continue;
138  }
139  // to keep track of processed CRUs
140  mProcessedCRUs[currentBuffer][relTF][cru] = true;
141 
142  // count total number of processed CRUs for given TF
143  ++mProcessedCRU[currentBuffer][relTF];
144 
145  const auto descr = tpcCRUHeader->dataDescription;
146  if (TPCFLPIDCDevice<TPCFLPIDCDeviceGroup>::getDataDescriptionIDCGroup() == descr) {
147  mIDCs[currentBuffer][cru][relTF] = pc.inputs().get<pmr::vector<float>>(ref);
148  LOGP(info, "receiving IDCs for CRU: {} of size {}", cru, mIDCs[currentBuffer][cru][relTF].size());
149  }
150  }
151  }
152 
153  LOGP(info, "number of received CRUs for current TF: {} Needed a total number of processed CRUs of: {} Current TF: {}", mProcessedCRU[currentBuffer][relTF], mCRUs.size(), tf);
154 
155  // check number of processed CRUs for previous TFs. If CRUs are missing for them, they are probably lost/not received
156  if (relTF == mTimeFrames - 1) {
157  for (int iTF = relTF - 1; iTF >= 0; --iTF) {
158  LOGP(info, "Checking rel TF: {} for missing CRUs", iTF);
159  if (mProcessedCRU[currentBuffer][iTF] != mCRUs.size()) {
160  LOGP(warning, "CRUs for TF: {} are missing!", iTF);
161 
162  // find actuall CRUs
163  for (auto& it : mProcessedCRUs[currentBuffer][iTF]) {
164  if (!it.second) {
165  LOGP(warning, "Couldnt find data for CRU {} possibly not received!", it.first);
166  }
167  }
168  }
169  }
170  }
171 
172  // check if all CRUs for current TF are already aggregated and send data
173  if ((mProcessedCRU[currentBuffer][relTF] == mCRUs.size() || mLoadFromFile) && !mDataSent[currentBuffer][relTF]) {
174  LOGP(info, "All data for current TF {} received. Sending data...", tf);
175  mDataSent[currentBuffer][relTF] = true;
176  ++mProcessedTFs[currentBuffer];
177  sendOutput(pc, currentOutLane, currentBuffer, relTF);
178  }
179 
180  if (mProcessedTFs[currentBuffer] == mTimeFrames) {
181  LOGP(info, "All TFs {} for current buffer received. Clearing buffer", tf);
182 
183  // resetting received CRUs
184  for (auto& crusMap : mProcessedCRUs[currentBuffer]) {
185  for (auto& it : crusMap) {
186  it.second = false;
187  }
188  }
189 
190  mProcessedTFs[currentBuffer] = 0; // reset processed TFs for next aggregation interval
191  std::fill(mProcessedCRU[currentBuffer].begin(), mProcessedCRU[currentBuffer].end(), 0);
192  std::fill(mDataSent[currentBuffer].begin(), mDataSent[currentBuffer].end(), false);
193 
194  // set integration range for next integration interval
195  mTFStart[mBuffer] = mTFEnd[!mBuffer] + 1;
196  mTFEnd[mBuffer] = mTFStart[mBuffer] + mTimeFrames - 1;
197 
198  // switch buffer
199  mBuffer = !mBuffer;
200 
201  // set output lane
202  mCurrentOutLane = ++mCurrentOutLane % mOutLanes;
203  }
204  }

236  }

245 DataProcessorSpec getTPCDistributeIDCSpec(const std::vector<uint32_t>& crus, const unsigned int timeframes, const unsigned int outlanes, const int firstTF, const bool loadFromFile)
246 {
247  std::vector<InputSpec> inputSpecs;
248  if (!loadFromFile) {
249  inputSpecs.emplace_back(InputSpec{"idcsgroup", ConcreteDataTypeMatcher{gDataOriginTPC, TPCFLPIDCDevice<TPCFLPIDCDeviceGroup>::getDataDescriptionIDCGroup()}, Lifetime::Timeframe});
250  }
251 
252  std::vector<OutputSpec> outputSpecs;
253  outputSpecs.reserve((outlanes + 1) * crus.size());
254  for (unsigned int lane = 0; lane < outlanes; ++lane) {
255  for (const auto cru : crus) {
256  const header::DataHeader::SubSpecificationType subSpec{cru + lane * CRU::MaxCRU};
257  outputSpecs.emplace_back(ConcreteDataMatcher{gDataOriginTPC, TPCDistributeIDCSpec::getDataDescriptionIDC(), subSpec});
258  }
259  }
260 
261  return DataProcessorSpec{
262  "tpc-distribute-idc",
263  inputSpecs,
264  outputSpecs,
265  AlgorithmSpec{adaptFromTask<TPCDistributeIDCSpec>(crus, timeframes, outlanes, loadFromFile, firstTF)}};
266 }
