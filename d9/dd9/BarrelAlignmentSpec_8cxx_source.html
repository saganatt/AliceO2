140 DataProcessorSpec getBarrelAlignmentSpec(GTrackID::mask_t src, DetID::mask_t dets)
141 {
142  std::vector<OutputSpec> outputs;
143  auto dataRequest = std::make_shared<DataRequest>();
144 
145  dataRequest->requestTracks(src, false);
146  dataRequest->requestClusters(src, false);
147  dataRequest->requestPrimaryVertertices(false);
148 
149  auto ccdbRequest = std::make_shared<o2::base::GRPGeomRequest>(true, // orbitResetTime
150  true, // GRPECS=true
151  false, // GRPLHCIF
152  false, // GRPMagField
153  false, // askMatLUT
154  o2::base::GRPGeomRequest::Alignments, // geometry
155  dataRequest->inputs);
156 
157  return DataProcessorSpec{
158  "barrel-alignment",
159  dataRequest->inputs,
160  outputs,
161  AlgorithmSpec{adaptFromTask<BarrelAlignmentSpec>(dataRequest, ccdbRequest, dets)},
162  Options{}};
163 }
