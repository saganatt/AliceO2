67  void init(InitContext& ic)
68  {
73  DigitIOBaseTask::init(ic);
74 
75  mNoFile = ic.options().get<bool>(OPTNAME_NO_FILE);
76  mBinary = not ic.options().get<bool>(OPTNAME_TXT);
77 
78  if (!mNoFile) {
79  auto outputFileName = ic.options().get<std::string>(OPTNAME_OUTFILE);
80  mStream = std::make_unique<std::fstream>(outputFileName, mBinary ? std::ios::out | std::ios::binary : std::ios::out);
81  if (mBinary) {
82  auto binaryFileFormat = ic.options().get<int>(OPTNAME_BINARY_FORMAT);
83  if (binaryFileFormat >= o2::mch::io::digitFileFormats.size()) {
84  throw std::invalid_argument(fmt::format("file version {} is unknown", binaryFileFormat));
85  }
86  auto maxsize = ic.options().get<int>(OPTNAME_MAX_SIZE);
87  LOGP(warn,
88  "Will dump binary information (version {}) up to a maximum size of {} KB",
89  binaryFileFormat, maxsize);
90  mWriter = std::make_unique<io::DigitWriter>(*mStream,
91  io::digitFileFormats[binaryFileFormat],
92  static_cast<size_t>(maxsize));
93  } else {
94  LOGP(warn, "Will dump textual information");
95  mWriter = std::make_unique<io::DigitWriter>(*mStream);
96  }
97  }
98  }
99 
100  void writeOrbits(gsl::span<const o2::mch::OrbitInfo> orbits)
101  {
102  if (orbits.size() && !mBinary) {
103  std::set<OrbitInfo> orderedOrbits(orbits.begin(), orbits.end());
104  for (auto o : orderedOrbits) {
105  (*mStream) << " FEEID " << o.getFeeID() << " LINK " << (int)o.getLinkID() << " ORBIT " << o.getOrbit() << std::endl;
106  }
107  }
108  }
109 
110  void write(gsl::span<const o2::mch::Digit> digits,
111  gsl::span<const o2::mch::ROFRecord> rofs,
112  gsl::span<const o2::mch::OrbitInfo> orbits)
113  {
114  if (mNoFile) {
115  return;
116  }
117  writeOrbits(orbits);
118  mWriter->write(digits, rofs);
119  }

121  void run(ProcessingContext& pc)
122  {
123  gsl::span<o2::mch::OrbitInfo> voidOrbitInfos;
124 
125  auto digits = pc.inputs().get<gsl::span<Digit>>("digits");
126  auto rofs = pc.inputs().get<gsl::span<o2::mch::ROFRecord>>("rofs");
127  auto orbits = mWithOrbits ? pc.inputs().get<gsl::span<o2::mch::OrbitInfo>>("orbits") : voidOrbitInfos;
128 
129  if (shouldProcess()) {
130  incNofProcessedTFs();
131  printSummary(digits, rofs, fmt::format("{:4d} orbits", orbits.size()).c_str());
132  printFull(digits, rofs);
133  write(digits, rofs, orbits);
134  }
135  incTFid();
136  }
137 
138  private:
139  std::unique_ptr<io::DigitWriter> mWriter = nullptr; // actual digit writer
140  std::unique_ptr<std::iostream> mStream = nullptr; // output stream
141  bool mNoFile = false; // disable output to file
142  bool mWithOrbits = false; // expect ORBITs as input (in addition to just digits)
143  bool mBinary = true; // output is a binary file
144 };

161 WorkflowSpec defineDataProcessing(const ConfigContext& cc)
162 {
163  WorkflowSpec specs;
164 
165  bool withOrbits = not cc.options().get<bool>(OPTNAME_WITHOUT_ORBITS);
166 
167  std::string input =
168  fmt::format("digits:MCH/{};rofs:MCH/{}",
169  cc.options().get<std::string>("input-digits-data-description"),
170  cc.options().get<std::string>("input-digitrofs-data-description"));
171  if (withOrbits) {
172  input += ";orbits:MCH/ORBITS";
173  }
174 
175  auto commonOptions = o2::mch::io::getCommonOptions();
176  auto options = Options{
177  {OPTNAME_OUTFILE, VariantType::String, "digits.out", {"output file name"}},
178  {OPTNAME_NO_FILE, VariantType::Bool, false, {"no output to file"}},
179  {OPTNAME_BINARY_FORMAT, VariantType::Int, 0, {"digit binary format to use"}},
180  {OPTNAME_TXT, VariantType::Bool, false, {"output digits in text format"}},
181  {OPTNAME_MAX_SIZE, VariantType::Int, std::numeric_limits<int>::max(), {"max output size (in KB)"}}};
182  options.insert(options.end(), commonOptions.begin(), commonOptions.end());
183 
184  Inputs inputs{o2::framework::select(input.c_str())};
185  const char* specName = "mch-digits-writer";
186 
187  if (!cc.options().get<bool>("enable-root-output")) {
188  DataProcessorSpec producer{
189  specName,
190  inputs,
191  Outputs{},
192  AlgorithmSpec{adaptFromTask<DigitsSinkTask>(withOrbits)},
193  options};
194  specs.push_back(producer);
195  } else {
196  auto rofs = std::find_if(inputs.begin(), inputs.end(), [](const InputSpec& is) { return is.binding == "rofs"; });
197  auto digits = std::find_if(inputs.begin(), inputs.end(), [](const InputSpec& is) { return is.binding == "digits"; });
198  DataProcessorSpec producer = MakeRootTreeWriterSpec(specName,
199  "mchdigits.root",
200  MakeRootTreeWriterSpec::TreeAttributes{"o2sim", "Tree MCH Digits"},
201  BranchDefinition<std::vector<ROFRecord>>{InputSpec{*rofs}, "rofs"},
202  BranchDefinition<std::vector<Digit>>{InputSpec{*digits}, "digits"})();
203  specs.push_back(producer);
204  }
205 
206  return specs;
207 }
