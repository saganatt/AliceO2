42 FCTLayer::FCTLayer(Int_t layerNumber, std::string layerName, Float_t z, Float_t rIn, Float_t rOut_SideL, Float_t Layerx2X0, Int_t type) : mLayerNumber(layerNumber), mLayerName(layerName), mx2X0(Layerx2X0), mType(type), mInnerRadius(rIn)
43 {
44  // Creates a simple parametrized FCT layer covering the given
45  // (rIn, rOut_SideL) range at the z layer position
46  mZ = -std::abs(z);
47  if (type == 0) { // Disk
48  mOuterRadius = rOut_SideL;
49  } else if (type == 1) { // Square
50  mSideLength = rOut_SideL;
51  }
52  Float_t Si_X0 = 9.37; // In cm
53  Float_t Pb_X0 = 0.5612; // In cm
54 
55  if (mType == 0) {
56  mChipThickness = Layerx2X0 * Si_X0;
57  LOG(info) << "Creating FCT Disk Layer " << mLayerNumber;
58  LOG(info) << " Using silicon X0 = " << Si_X0 << " to emulate layer radiation length.";
59  LOG(info) << " Layer z = " << mZ << " ; R_in = " << mInnerRadius << " ; R_out = " << mOuterRadius << " ; x2X0 = " << mx2X0 << " ; ChipThickness = " << mChipThickness;
60  } else if (mType == 1) {
61  mChipThickness = Layerx2X0 * Si_X0;
62  LOG(info) << "Creating FCT Square Layer " << mLayerNumber;
63  LOG(info) << " Using silicon X0 = " << Si_X0 << " to emulate layer radiation length.";
64  LOG(info) << " Layer z = " << mZ << " ; R_in = " << mInnerRadius << " ; L_side = " << mSideLength << " ; x2X0 = " << mx2X0 << " ; ChipThickness = " << mChipThickness;
65  } else if (mType == 2) {
66  mChipThickness = Layerx2X0 * Pb_X0;
67  LOG(info) << "Creating FCT Converter Layer " << mLayerNumber;
68  LOG(info) << " Using lead X0 = " << Pb_X0 << " to emulate layer radiation length.";
69  LOG(info) << " Layer z = " << mZ << " ; R_in = " << mInnerRadius << " ; R_out = " << mOuterRadius << " ; x2X0 = " << mx2X0 << " ; ChipThickness = " << mChipThickness;
70  }
71 }

73 void FCTLayer::createLayer(TGeoVolume* motherVolume)
74 {
75  if (mType == 0) {
76  createDiskLayer(motherVolume);
77  } else if (mType == 1) {
78  createSquareLayer(motherVolume);

85 void FCTLayer::createDiskLayer(TGeoVolume* motherVolume)
86 {
87  if (mLayerNumber < 0) {
88  return;
89  }
90  // Create tube, set sensitive volume, add to mother volume
91 
92  std::string chipName = o2::fct::GeometryTGeo::getFCTChipPattern() + std::to_string(mLayerNumber),
93  sensName = Form("%s_%d", GeometryTGeo::getFCTSensorPattern(), mLayerNumber);
94  TGeoTube* sensor = new TGeoTube(mInnerRadius, mOuterRadius, mChipThickness / 2);
95  TGeoTube* chip = new TGeoTube(mInnerRadius, mOuterRadius, mChipThickness / 2);
96  TGeoTube* layer = new TGeoTube(mInnerRadius, mOuterRadius, mChipThickness / 2);
97 
98  TGeoMedium* medSi = gGeoManager->GetMedium("FCT_SI$");
99  TGeoMedium* medAir = gGeoManager->GetMedium("FCT_AIR$");
100 
101  TGeoVolume* sensVol = new TGeoVolume(sensName.c_str(), sensor, medSi);
102  TGeoVolume* chipVol = new TGeoVolume(chipName.c_str(), chip, medSi);
103  TGeoVolume* layerVol = new TGeoVolume(mLayerName.c_str(), layer, medAir);
104 
105  LOG(info) << "Inserting " << sensVol->GetName() << " inside " << chipVol->GetName();
106  chipVol->AddNode(sensVol, 1, nullptr);
107 
108  LOG(info) << "Inserting " << chipVol->GetName() << " inside " << layerVol->GetName();
109  layerVol->AddNode(chipVol, 1, nullptr);
110 
111  // Finally put everything in the mother volume
112  auto FwdDiskRotation = new TGeoRotation("FwdDiskRotation", 0, 0, 180);
113  auto FwdDiskCombiTrans = new TGeoCombiTrans(0, 0, mZ, FwdDiskRotation);
114 
115  LOG(info) << "Inserting " << layerVol->GetName() << " inside " << motherVolume->GetName();
116  motherVolume->AddNode(layerVol, 1, FwdDiskCombiTrans);
117 }
