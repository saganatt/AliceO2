54 WorkflowSpec defineDataProcessing(ConfigContext const& config)
55 {
56  using namespace o2::tpc;
57 
58  const std::string sgroupPads = config.options().get<std::string>("groupPads");
59  const std::string sgroupRows = config.options().get<std::string>("groupRows");
60  const std::string sgroupLastRowsThreshold = config.options().get<std::string>("groupLastRowsThreshold");
61  const std::string sgroupLastPadsThreshold = config.options().get<std::string>("groupLastPadsThreshold");
62  ParameterIDCGroup::setGroupingParameterFromString(sgroupPads, sgroupRows, sgroupLastRowsThreshold, sgroupLastPadsThreshold);
63 
64  // set up configuration
65  o2::conf::ConfigurableParam::updateFromFile(config.options().get<std::string>("configFile"));
66  o2::conf::ConfigurableParam::updateFromString(config.options().get<std::string>("configKeyValues"));
67  o2::conf::ConfigurableParam::writeINI("o2tpcfactorizeidc_configuration.ini");
68 
69  const auto tpcCRUs = o2::RangeTokenizer::tokenize<int>(config.options().get<std::string>("crus"));
70  const auto nCRUs = tpcCRUs.size();
71  const auto timeframes = static_cast<unsigned int>(config.options().get<int>("timeframes"));
72  const auto timeframesDeltaIDC = static_cast<unsigned int>(config.options().get<int>("timeframesDeltaIDC"));
73  const auto debug = config.options().get<bool>("debug");
74  const auto groupIDCs = config.options().get<bool>("groupIDCs");
75  const auto sendOutput = config.options().get<bool>("sendOutput");
76  const auto sendOutputFFT = config.options().get<bool>("sendOutputFFT");
77  const auto nthreadsFactorization = static_cast<unsigned long>(config.options().get<int>("nthreads-IDC-factorization"));
78  IDCFactorization::setNThreads(nthreadsFactorization);
79  const auto nthreadsGrouping = static_cast<unsigned long>(config.options().get<int>("nthreads-grouping"));
80  IDCAverageGroup<IDCAverageGroupTPC>::setNThreads(nthreadsGrouping);
81  const auto nLanes = static_cast<unsigned int>(config.options().get<int>("input-lanes"));
82  const bool usePrecisetimeStamp = !config.options().get<bool>("use-approximate-timestamp");
83 
84  const int compressionTmp = config.options().get<int>("compression");
85  IDCDeltaCompression compression;
86  switch (compressionTmp) {
87  case static_cast<int>(IDCDeltaCompression::NO):
88  case static_cast<int>(IDCDeltaCompression::MEDIUM):
89  case static_cast<int>(IDCDeltaCompression::HIGH):
90  compression = static_cast<IDCDeltaCompression>(compressionTmp);
91  break;
92  default:
93  LOGP(error, "wrong compression type set. Setting compression to medium compression");
94  compression = static_cast<IDCDeltaCompression>(IDCDeltaCompression::MEDIUM);
95  break;
96  }
97 
98  const auto first = tpcCRUs.begin();
99  const auto last = std::min(tpcCRUs.end(), first + nCRUs);
100  const std::vector<uint32_t> rangeCRUs(first, last);
101 
102  WorkflowSpec workflow;
103  workflow.reserve(nLanes);
104  for (int ilane = 0; ilane < nLanes; ++ilane) {
105  groupIDCs ? workflow.emplace_back(getTPCFactorizeIDCSpec<TPCFactorizeIDCSpecGroup>(ilane, rangeCRUs, timeframes, timeframesDeltaIDC, compression, debug, sendOutput, usePrecisetimeStamp, sendOutputFFT)) : workflow.emplace_back(getTPCFactorizeIDCSpec<TPCFactorizeIDCSpecNoGroup>(ilane, rangeCRUs, timeframes, timeframesDeltaIDC, compression, debug, sendOutput, usePrecisetimeStamp, sendOutputFFT));
106  }
107  return workflow;
108 }
