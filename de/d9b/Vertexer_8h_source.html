154 template <typename... T>
155 float Vertexer::evaluateTask(void (Vertexer::*task)(T...), bool verbose, const char* taskName, std::function<void(std::string s)> logger,
156  T&&... args)
157 {
158  float diff{0.f};
159 
160  if (constants::DoTimeBenchmarks) {
161  auto start = std::chrono::high_resolution_clock::now();
162  (this->*task)(std::forward<T>(args)...);
163  auto end = std::chrono::high_resolution_clock::now();
164 
165  std::chrono::duration<double, std::milli> diff_t{end - start};
166  diff = diff_t.count();
167 
168  std::stringstream sstream;
169  if (taskName == nullptr) {
170  sstream << diff << "\t";
171  } else {
172  sstream << std::setw(2) << " - " << taskName << " completed in: " << diff << " ms";
173  }
174  if (verbose) {
175  logger(sstream.str());
176  }
177  } else {
178  (this->*task)(std::forward<T>(args)...);
179  }
180 
181  return diff;
182 }
