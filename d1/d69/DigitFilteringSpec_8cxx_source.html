38 class DigitFilteringTask
39 {
40  public:
41  DigitFilteringTask(bool useMC) : mUseMC{useMC} {}
42 
43  void init(InitContext& ic)
44  {
45  mSanityCheck = DigitFilterParam::Instance().sanityCheck;
46  int minADC = DigitFilterParam::Instance().minADC;
47  bool rejectBackground = DigitFilterParam::Instance().rejectBackground;
48  if (mUseMC) {
49  rejectBackground = false;
50  LOGP(warn, "background rejection not tuned for MC so far : disabling it");
51  }
52  mIsGoodDigit = createDigitFilter(minADC, rejectBackground, false);
53  // at digit filtering stage it is important to keep the 3rd parameter
54  // to false in the call above : the idea is to not cut too much
55  // on the tails of the charge distributions otherwise the clustering
56  // resolution will suffer.
57  // That's why we only apply the "reject background" filter, which
58  // is a loose background cut that does not penalize the signal
59 
60  mTimeCalib = DigitFilterParam::Instance().timeOffset;
61  }
62 
63  void shiftDigitsTime(gsl::span<ROFRecord> rofs, gsl::span<Digit> digits)
64  {
65  for (auto i = 0; i < rofs.size(); i++) {
66  ROFRecord& rof = rofs[i];
67  rof.getBCData() += mTimeCalib;
68  }
69 
70  for (auto i = 0; i < digits.size(); i++) {
71  Digit& d = digits[i];
72  d.setTime(d.getTime() + mTimeCalib);
73  }
74  }
75 
76  void run(ProcessingContext& pc)
77  {
78  // get input
79  auto iRofs = pc.inputs().get<gsl::span<ROFRecord>>("rofs");
80  auto iDigits = pc.inputs().get<gsl::span<Digit>>("digits");
81  auto iLabels = mUseMC ? pc.inputs().get<MCTruthContainer<MCCompLabel>*>("labels") : nullptr;
82 
83  bool abort{false};
84 
85  if (mSanityCheck) {
86  LOGP(info, "performing sanity checks");
87  auto error = sanityCheck(iRofs, iDigits);
88 
89  if (!isOK(error)) {
90  if (error.nofOutOfBounds > 0) {
91  LOGP(error, asString(error));
92  LOGP(error, "in a TF with {} rofs and {} digits", iRofs.size(), iDigits.size());
93  abort = true;
94  }
95  }
96  }
97 
98  // create the output messages
99  auto& oRofs = pc.outputs().make<std::vector<ROFRecord>>(OutputRef{"rofs"});
100  auto& oDigits = pc.outputs().make<std::vector<Digit>>(OutputRef{"digits"});
101  auto oLabels = mUseMC ? &pc.outputs().make<MCTruthContainer<MCCompLabel>>(OutputRef{"labels"}) : nullptr;
102 
103  if (!abort) {
104  int cursor{0};
105  for (const auto& irof : iRofs) {
106  const auto digits = iDigits.subspan(irof.getFirstIdx(), irof.getNEntries());
107 
108  // filter the digits from the current ROF
109  for (auto i = 0; i < digits.size(); i++) {
110  const auto& d = digits[i];
111  if (mIsGoodDigit(d)) {
112  oDigits.emplace_back(d);
113  if (iLabels) {
114  oLabels->addElements(oLabels->getIndexedSize(), iLabels->getLabels(i + irof.getFirstIdx()));
115  }
116  }
117  }
118  int nofGoodDigits = oDigits.size() - cursor;
119  if (nofGoodDigits > 0) {
120  // we create an ouput ROF only if at least one digit from
121  // the input ROF passed the filtering
122  oRofs.emplace_back(ROFRecord(irof.getBCData(),
123  cursor,
124  nofGoodDigits,
125  irof.getBCWidth()));
126  cursor += nofGoodDigits;
127  }
128  }
129  }
130 
131  auto labelMsg = mUseMC ? fmt::format("| {} labels (out of {})", oLabels->getNElements(), iLabels->getNElements()) : "";
132 
133  LOGP(info, "Kept after filtering : {} rofs (out of {}) | {} digits (out of {}) {}\n",
134  oRofs.size(), iRofs.size(),
135  oDigits.size(), iDigits.size(),
136  labelMsg);
137 
138  if (mTimeCalib != 0) {
139  shiftDigitsTime(oRofs, oDigits);
140  }
141 
142  if (abort) {
143  LOGP(error, "Sanity check failed");
144  }
145  }
146 
147  private:
148  bool mSanityCheck;
149  bool mUseMC;
150  DigitFilter mIsGoodDigit;
151  int32_t mTimeCalib{0};
152 };

178  output += fmt::format(";labels:MCH/{}/0", outputDigitLabelDataDescription);

182  auto matchers = select(output.c_str());
183  for (auto& matcher : matchers) {
184  outputs.emplace_back(DataSpecUtils::asOutputSpec(matcher));
185  }
186 
187  return DataProcessorSpec{
188  specName.data(),
189  Inputs{select(input.c_str())},
190  outputs,
191  AlgorithmSpec{adaptFromTask<DigitFilteringTask>(useMC)},
