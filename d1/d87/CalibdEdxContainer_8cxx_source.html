193 CalDet<float> CalibdEdxContainer::processThresholdMap(const CalDet<float>& thresholdMap, const float maxThreshold, const int nPadsInRowCl, const int nPadsInPadCl) const
194 {
195  CalDet<float> thresholdMapProcessed(thresholdMap);
196 
197  for (unsigned int sector = 0; sector < Mapper::NSECTORS; ++sector) {
198  for (unsigned int region = 0; region < Mapper::NREGIONS; ++region) {
199  const int maxRow = Mapper::ROWSPERREGION[region] - 1;
200  for (int lrow = 0; lrow <= maxRow; ++lrow) {
201  // find first row of the cluster
202  const int rowStart = std::clamp(lrow - nPadsInRowCl, 0, maxRow);
203  const int rowEnd = std::clamp(lrow + nPadsInRowCl, 0, maxRow);
204  const int addPadsStart = Mapper::ADDITIONALPADSPERROW[region][lrow];
205 
206  for (unsigned int pad = 0; pad < Mapper::PADSPERROW[region][lrow]; ++pad) {
207  float sumThr = 0;
208  int countThr = 0;
209  // loop ove the rows from the cluster
210  for (int rowCl = rowStart; rowCl <= rowEnd; ++rowCl) {
211  // shift local pad in row in case current row from the cluster has more pads in the row
212  const int addPadsCl = Mapper::ADDITIONALPADSPERROW[region][rowCl];
213  const int diffAddPads = addPadsCl - addPadsStart;
214  const int padClCentre = pad + diffAddPads;
215 
216  const int maxPad = Mapper::PADSPERROW[region][rowCl] - 1;
217  const int padStart = std::clamp(padClCentre - nPadsInPadCl, 0, maxPad);
218  const int padEnd = std::clamp(padClCentre + nPadsInPadCl, 0, maxPad);
219  for (int padCl = padStart; padCl <= padEnd; ++padCl) {
220  const int globalPad = Mapper::getGlobalPadNumber(rowCl, padCl, region);
221  // skip for current cluster position as the charge there is not effected from the thresold
222  if (padCl == pad && rowCl == lrow) {
223  continue;
224  }
225 
226  float threshold = thresholdMap.getValue(sector, globalPad);
227  if (threshold > maxThreshold) {
228  threshold = maxThreshold;
229  }
230 
231  sumThr += threshold;
232  ++countThr;
233  }
234  }
235  const float meanThresold = sumThr / countThr;
236  const int globalPad = Mapper::getGlobalPadNumber(lrow, pad, region);
237  thresholdMapProcessed.setValue(sector, globalPad, meanThresold);
238  }
239  }
240  }
241  }
242  return thresholdMapProcessed;
243 }

257 void CalibdEdxContainer::setDefaultZeroSupresssionThreshold()
258 {
259  const float defaultVal = getMinZeroSupresssionThreshold() + (getMaxZeroSupresssionThreshold() - getMinZeroSupresssionThreshold()) / 2;
260  mThresholdMap.setMinCorrectionFactor(defaultVal - 0.1f);
261  mThresholdMap.setMaxCorrectionFactor(defaultVal + 0.1f);
262  for (int sector = 0; sector < o2::tpc::constants::MAXSECTOR; ++sector) {
263  for (unsigned short globPad = 0; globPad < TPC_PADS_IN_SECTOR; ++globPad) {
264  mThresholdMap.setGainCorrection(sector, globPad, defaultVal);
265  }
266  }

279 {

283  const std::size_t flatbufferSize = calibTrackTopologyTmp.getFlatBufferSize();

287

290  FlatObject::finishConstruction(totalSize);
291

293  obj = reinterpret_cast<Type*>(mFlatBufferPtr);
294 
295  // deep copy of CalibdEdxTrackTopologyPol to buffer without moving the flat buffer to correct address
296  obj->cloneFromObject(calibTrackTopologyTmp, nullptr);
297 
298  // seting the buffer of the splines to current buffer
299  obj->moveBufferTo(objSize + mFlatBufferPtr);
