28 void RawBufferContext::clear()
29 {
30  // On send we move the header, but the payload remains
31  // there because what's really sent is the copy of the raw
32  // payload will be cleared by the mMessages.clear()
33  for (auto& m : mMessages) {
34  assert(m.header == nullptr);
35  // NOTE: payloads can be empty so m.payload == nullptr should
36  // be an actual issue.
37  assert(m.payload != nullptr);
38  m.destroyPayload();
39  m.payload = nullptr;
40  }
41 
42  mMessages.clear();
43 }
