30 uint64_t processing_helpers::getRunNumber(ProcessingContext& pc)
31 {
32  const std::string NAStr = "NA";
33 
34  uint64_t run = 0;
35  const auto dh = DataRefUtils::getHeader<o2::header::DataHeader*>(pc.inputs().getFirstValid(true));
36  if (dh->runNumber != 0) {
37  run = dh->runNumber;
38  }
39  // check runNumber with FMQ property, if set, override DH number
40  {
41  auto runNStr = pc.services().get<RawDeviceService>().device()->fConfig->GetProperty<std::string>("runNumber", NAStr);
42  if (runNStr != NAStr) {
43  size_t nc = 0;
44  long runNProp = 0;
45  try {
46  runNProp = std::stol(runNStr, &nc);
47  } catch (...) {
48  nc = (size_t)-1; // makes the next check fail if stol throws when it cannot parse the number
49  }
50  if (nc != runNStr.size()) {
51  LOGP(error, "Property runNumber={} is provided but is not a number, ignoring", runNStr);
52  } else {
53  run = runNProp;
54  }
55  }
56  }
57 
58  return run;
59 }
