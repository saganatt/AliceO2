55 class TimeClusterFinderTask

57  public:
58  //_________________________________________________________________________________________________
59  void init(framework::InitContext& ic)
60  {
61  const auto& param = TimeClusterizerParam::Instance();
62  mTimeClusterWidth = param.maxClusterWidth;
63  mNbinsInOneWindow = param.peakSearchNbins;
64  mMinDigitPerROF = param.minDigitsPerROF;
65  mOnlyTrackable = param.onlyTrackable;
66  mPeakSearchSignalOnly = param.peakSearchSignalOnly;
67  mIRFramesOnly = param.irFramesOnly;
68  mDebug = ic.options().get<bool>("mch-debug");
69 
70  if (mDebug) {
71  fair::Logger::SetConsoleColor(true);
72  }
73  // number of bins must be >= 3
74  if (mNbinsInOneWindow < 3) {
75  mNbinsInOneWindow = 3;
76  }
77  // number of bins must be odd
78  if ((mNbinsInOneWindow % 2) == 0) {
79  mNbinsInOneWindow += 1;
80  }
81 
82  LOGP(info, "TimeClusterWidth : {}", mTimeClusterWidth);
83  LOGP(info, "BinsInOneWindow : {} ", mNbinsInOneWindow);
84  LOGP(info, "MinDigitPerROF : {}", mMinDigitPerROF);
85  LOGP(info, "OnlyTrackable : {}", mOnlyTrackable);
86  LOGP(info, "PeakSearchSignalOnly: {}", mPeakSearchSignalOnly);
87  LOGP(info, "IRFramesOnly : {}", mIRFramesOnly);
88 
89  auto stop = [this]() {
90  if (mTFcount) {
91  LOGP(info, "\nduration = {} us / TF\n", mTimeProcess.count() * 1000 / mTFcount);
92  }
93  };
94  ic.services().get<CallbackService>().set(CallbackService::Id::Stop, stop);
95  }
96 
97  //_________________________________________________________________________________________________
98  void run(framework::ProcessingContext& pc)
99  {
100  auto rofs = pc.inputs().get<gsl::span<o2::mch::ROFRecord>>("rofs");
101  auto digits = pc.inputs().get<gsl::span<o2::mch::Digit>>("digits");
102 
103  o2::mch::ROFTimeClusterFinder rofProcessor(rofs, digits, mTimeClusterWidth, mNbinsInOneWindow, mPeakSearchSignalOnly, mDebug);
104 
105  if (mDebug) {
106  LOGP(warning, "{:=>60} ", fmt::format("{:6d} Input ROFS", rofs.size()));
107  }
108 
109  auto tStart = std::chrono::high_resolution_clock::now();
110  rofProcessor.process();
111  auto tEnd = std::chrono::high_resolution_clock::now();
112  mTimeProcess += tEnd - tStart;
113 
114  if (mDebug) {
115  LOGP(warning, "{:=>60} ", fmt::format("{:6d} Output ROFS", rofProcessor.getROFRecords().size()));
116  }
117 
118  auto& outRofs = pc.outputs().make<std::vector<ROFRecord>>(OutputRef{"rofs"});
119  const auto& pRofs = rofProcessor.getROFRecords();
120 
121  // prepare the list of filters we want to apply to ROFs
122  std::vector<ROFFilter> filters;
123 
124  if (mOnlyTrackable) {
125  // selects only ROFs that are trackable
126  const auto& trackerParam = TrackerParam::Instance();
127  std::array<bool, 5> requestStation{
128  trackerParam.requestStation[0],
129  trackerParam.requestStation[1],
130  trackerParam.requestStation[2],
131  trackerParam.requestStation[3],
132  trackerParam.requestStation[4]};
133  filters.emplace_back(createTrackableFilter(digits,
134  requestStation,
135  trackerParam.moreCandidates));
136  }
137  if (mMinDigitPerROF > 0) {
138  // selects only those ROFs have that minimum number of digits
139  filters.emplace_back(createMultiplicityFilter(mMinDigitPerROF));
140  }
141  if (mIRFramesOnly) {
142  // selects only those ROFs that overlop some IRFrame
143  auto irFrames = pc.inputs().get<gsl::span<o2::dataformats::IRFrame>>("irframes");
144  filters.emplace_back(createIRFrameFilter(irFrames));
145  }
146 
147  // a single filter which is the AND combination of the elements of the filters vector
148  auto filter = createROFFilter(filters);
149 
150  std::copy_if(begin(pRofs),
151  end(pRofs),
152  std::back_inserter(outRofs),
153  filter);
154 
155  const float p1 = rofs.size() > 0 ? 100. * pRofs.size() / rofs.size() : 0;
156  const float p2 = rofs.size() > 0 ? 100. * outRofs.size() / rofs.size() : 0;
157 
158  LOGP(info,
159  "TF {} Processed {} input ROFs, "
160  "time-clusterized them into {} ROFs ({:3.0f}%) "
161  "and output {} ({:3.0f}%) of them",
162  mTFcount, rofs.size(),
163  pRofs.size(), p1,
164  outRofs.size(), p2);
165  mTFcount += 1;
166  }
167 
168  private:
169  std::chrono::duration<double, std::milli>
170  mTimeProcess{};
171 
172  uint32_t mTimeClusterWidth;
173  uint32_t mNbinsInOneWindow;
174  int mTFcount{0};
175  int mDebug{0};
176  int mMinDigitPerROF;
177  bool mPeakSearchSignalOnly;
178  bool mOnlyTrackable;
179  bool mIRFramesOnly;
180 };

183 o2::framework::DataProcessorSpec
184  getTimeClusterFinderSpec(const char* specName,
185  std::string_view inputDigitDataDescription,
186  std::string_view inputDigitRofDataDescription,
187  std::string_view outputDigitRofDataDescription,
188  std::string_view inputIRFrameDataDescription)
189 {
190  std::string input = fmt::format("rofs:MCH/{}/0;digits:MCH/{}/0",
191  inputDigitRofDataDescription.data(),
192  inputDigitDataDescription.data());
193  if (TimeClusterizerParam::Instance().irFramesOnly && inputIRFrameDataDescription.size()) {
194  LOGP(info, "will select IRFrames from {}", inputIRFrameDataDescription);
195  input += ";irframes:";
196  input += inputIRFrameDataDescription;
197  }
198  std::string output = fmt::format("rofs:MCH/{}/0", outputDigitRofDataDescription.data());
199 
200  std::vector<OutputSpec> outputs;
201  auto matchers = select(output.c_str());
202  for (auto& matcher : matchers) {
203  outputs.emplace_back(DataSpecUtils::asOutputSpec(matcher));
204  }
205 
206  return DataProcessorSpec{
207  specName,
208  Inputs{select(input.c_str())},
209  outputs,
210  AlgorithmSpec{adaptFromTask<TimeClusterFinderTask>()},
211  Options{{"mch-debug", VariantType::Bool, false, {"enable verbose output"}}}};
212 }
