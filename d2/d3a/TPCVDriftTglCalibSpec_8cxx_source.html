72 void TPCVDriftTglCalibSpec::sendOutput(DataAllocator& output)
73 {
74  // extract CCDB infos and calibration objects, convert it to TMemFile and send them to the output
75  // TODO in principle, this routine is generic, can be moved to Utils.h
76  using clbUtils = o2::calibration::Utils;
77  const auto& payloadVec = mCalibrator->getVDPerSlot();
78  auto& infoVec = mCalibrator->getCCDBInfoPerSlot(); // use non-const version as we update it
79  assert(payloadVec.size() == infoVec.size());
80 
81  for (uint32_t i = 0; i < payloadVec.size(); i++) {
82  auto& w = infoVec[i];
83  auto image = o2::ccdb::CcdbApi::createObjectImage(&payloadVec[i], &w);
84  LOG(info) << "Sending object " << w.getPath() << "/" << w.getFileName() << " of size " << image->size()
85  << " bytes, valid for " << w.getStartValidityTimestamp() << " : " << w.getEndValidityTimestamp();
86  output.snapshot(Output{o2::calibration::Utils::gDataOriginCDBPayload, "TPCVDTGL", i, Lifetime::Sporadic}, *image.get()); // vector<char>
87  output.snapshot(Output{o2::calibration::Utils::gDataOriginCDBWrapper, "TPCVDTGL", i, Lifetime::Sporadic}, w); // root-serialized
88  }
89  if (payloadVec.size()) {
90  mCalibrator->initOutput(); // reset the outputs once they are already sent
91  }
92 }
