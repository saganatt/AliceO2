86  size_t getNGoodEntries() const
87  {
88  size_t posGood = 0;
89  while (posGood < fes.size()) {
90  if (!fes[posGood].all()) {
91  break;
92  }
93  ++posGood;
94  }
95  return posGood;
96  }
97 
99  gsl::span<const DataPoint> getGoodData() const
100  {
101  return gsl::span<const DataPoint>(data.data(), data.data() + getNGoodEntries());
102  }
103 
105  void clearGoodData()
106  {
107  const auto pos = getNGoodEntries();
108  data.erase(data.begin(), data.begin() + pos);
109  fes.erase(fes.begin(), fes.begin() + pos);
