82 #define MONITORING_QUEUE_SIZE 100
83 o2::framework::ServiceSpec CommonServices::monitoringSpec()
84 {
85  return ServiceSpec{
86  .name = "monitoring",
87  .init = [](ServiceRegistry& registry, DeviceState&, fair::mq::ProgOptions& options) -> ServiceHandle {
88  void* service = nullptr;
89  bool isWebsocket = strncmp(options.GetPropertyAsString("driver-client-backend").c_str(), "ws://", 4) == 0;
90  bool isDefault = options.GetPropertyAsString("monitoring-backend") == "default";
91  bool useDPL = (isWebsocket && isDefault) || options.GetPropertyAsString("monitoring-backend") == "dpl://";
92  o2::monitoring::Monitoring* monitoring;
93  if (useDPL) {
94  monitoring = new Monitoring();
95  auto dplBackend = std::make_unique<DPLMonitoringBackend>(registry);
96  (dynamic_cast<o2::monitoring::Backend*>(dplBackend.get()))->setVerbosity(o2::monitoring::Verbosity::Debug);
97  monitoring->addBackend(std::move(dplBackend));
98  } else {
99  auto backend = isDefault ? "infologger://" : options.GetPropertyAsString("monitoring-backend");
100  monitoring = MonitoringFactory::Get(backend).release();
101  }
102  service = monitoring;
103  monitoring->enableBuffering(MONITORING_QUEUE_SIZE);
104  assert(registry.get<DeviceSpec const>().name.empty() == false);
105  monitoring->addGlobalTag("dataprocessor_id", registry.get<DeviceSpec const>().id);
106  monitoring->addGlobalTag("dataprocessor_name", registry.get<DeviceSpec const>().name);
107  monitoring->addGlobalTag("dpl_instance", options.GetPropertyAsString("shm-segment-id"));
108  return ServiceHandle{TypeIdHelpers::uniqueId<Monitoring>(), service};
109  },
110  .configure = noConfiguration(),
111  .start = [](ServiceRegistry& services, void* service) {
112  o2::monitoring::Monitoring* monitoring = (o2::monitoring::Monitoring *) service;
113  auto& context = services.get<DataTakingContext>();
114 
115  try {
116  monitoring->setRunNumber(std::stoul(context.runNumber.c_str()));
117  } catch (...) {
118  } },
119  .exit = [](ServiceRegistry& registry, void* service) {
120  Monitoring* monitoring = reinterpret_cast<Monitoring*>(service);
121  delete monitoring; },
122  .kind = ServiceKind::Serial};
123 }

147 o2::framework::ServiceSpec CommonServices::datatakingContextSpec()
148 {
149  return ServiceSpec{
150  .name = "datataking-contex",
151  .init = simpleServiceInit<DataTakingContext, DataTakingContext>(),
152  .configure = noConfiguration(),
153  .preProcessing = [](ProcessingContext& processingContext, void* service) {
154  auto& context = processingContext.services().get<DataTakingContext>();
155  // Only on the first message
156  if (context.source == OrbitResetTimeSource::Data) {
157  return;
158  }
159  // Only if we do not have already the proper number from CTP
160  if (context.source == OrbitResetTimeSource::CTP) {
161  return;
162  }
163  context.source = OrbitResetTimeSource::Data;
164  context.orbitResetTime = -1;
165  for (auto const& ref : processingContext.inputs()) {
166  const o2::framework::DataProcessingHeader *dph = o2::header::get<DataProcessingHeader*>(ref.header);
167  const auto* dh = o2::header::get<o2::header::DataHeader*>(ref.header);
168  if (!dph || !dh) {
169  continue;
170  }
171  LOGP(debug, "Orbit reset time from data: {} ", dph->creation);
172  context.orbitResetTime = dph->creation;
173  context.runNumber = fmt::format("{}", dh->runNumber);
174  break;
175  } },
176  .start = [](ServiceRegistry& services, void* service) {
177  auto& context = services.get<DataTakingContext>();
178  auto extRunNumber = services.get<RawDeviceService>().device()->fConfig->GetProperty<std::string>("runNumber", "unspecified");
179  if (extRunNumber != "unspecified" || context.runNumber == "0") {
180  context.runNumber = extRunNumber;
181  }
182  auto extLHCPeriod = services.get<RawDeviceService>().device()->fConfig->GetProperty<std::string>("lhc_period", "unspecified");
183  if (extLHCPeriod != "unspecified") {
184  context.lhcPeriod = extLHCPeriod;
185  } else {
186  static const char* months[12] = {"JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"};
187  time_t now = time(nullptr);
188  auto ltm = gmtime(&now);
189  context.lhcPeriod = months[ltm->tm_mon];
190  LOG(info) << "LHCPeriod is not available, using current month " << context.lhcPeriod;
191  }
192 
193  auto extRunType = services.get<RawDeviceService>().device()->fConfig->GetProperty<std::string>("run_type", "unspecified");
194  if (extRunType != "unspecified") {
195  context.runType = extRunType;
196  }
197  auto extEnvId = services.get<RawDeviceService>().device()->fConfig->GetProperty<std::string>("environment_id", "unspecified");
198  if (extEnvId != "unspecified") {
199  context.envId = extEnvId;
200  }
201  auto extDetectors = services.get<RawDeviceService>().device()->fConfig->GetProperty<std::string>("detectors", "unspecified");
202  if (extDetectors != "unspecified") {
203  context.detectors = extDetectors;
204  }
205  // FIXME: we actually need to get the orbit, not only to know where it is
206  std::string orbitResetTimeUrl = services.get<RawDeviceService>().device()->fConfig->GetProperty<std::string>("orbit-reset-time", "ccdb://CTP/Calib/OrbitResetTime");
207  auto is_number = [](const std::string& s) -> bool {
208  return !s.empty() && std::all_of(s.begin(), s.end(), ::isdigit);
209  };
210 
211  if (orbitResetTimeUrl.rfind("file://") == 0) {
212  // FIXME: read it from a file
213  context.orbitResetTime = 490917600;
214  } else if (orbitResetTimeUrl.rfind("http://") == 0) {
215  // FIXME: read it from ccdb
216  context.orbitResetTime = 490917600;
217  } else if (is_number(orbitResetTimeUrl)) {
218  context.orbitResetTime = std::stoull(orbitResetTimeUrl.data());
219  // FIXME: specify it from the command line
220  } else {
221  context.orbitResetTime = 490917600;
222  }
223  context.nOrbitsPerTF = services.get<RawDeviceService>().device()->fConfig->GetProperty<uint64_t>("Norbits_per_TF", 128); },
224  .kind = ServiceKind::Serial};
225 }

243 // Creates the sink for FairLogger / InfoLogger integration
244 auto createInfoLoggerSinkHelper(InfoLogger* logger, InfoLoggerContext* ctx)
245 {
246  return [logger,
247  ctx](const std::string& content, const fair::LogMetaData& metadata) {
248  // translate FMQ metadata
249  InfoLogger::InfoLogger::Severity severity = InfoLogger::Severity::Undefined;
250  int level = InfoLogger::undefinedMessageOption.level;
251 
252  if (metadata.severity_name == fair::Logger::SeverityName(fair::Severity::nolog)) {
253  // discard
254  return;
255  } else if (metadata.severity_name == fair::Logger::SeverityName(fair::Severity::fatal)) {
256  severity = InfoLogger::Severity::Fatal;
257  level = 1;
258  } else if (metadata.severity_name == fair::Logger::SeverityName(fair::Severity::error)) {
259  severity = InfoLogger::Severity::Error;
260  level = 3;
261  } else if (metadata.severity_name == fair::Logger::SeverityName(fair::Severity::alarm)) {
262  severity = InfoLogger::Severity::Warning;
263  level = 4;
264  } else if (metadata.severity_name == fair::Logger::SeverityName(fair::Severity::important)) {
265  severity = InfoLogger::Severity::Info;
266  level = 5;
267  } else if (metadata.severity_name == fair::Logger::SeverityName(fair::Severity::warn)) {
268  severity = InfoLogger::Severity::Warning;
269  level = 6;
270  } else if (metadata.severity_name == fair::Logger::SeverityName(fair::Severity::state)) {
271  severity = InfoLogger::Severity::Info;
272  level = 8;
273  } else if (metadata.severity_name == fair::Logger::SeverityName(fair::Severity::info)) {
274  severity = InfoLogger::Severity::Info;
275  level = 10;
276  } else if (metadata.severity_name == fair::Logger::SeverityName(fair::Severity::debug)) {
277  severity = InfoLogger::Severity::Debug;
278  level = 11;
279  } else if (metadata.severity_name == fair::Logger::SeverityName(fair::Severity::debug1)) {
280  severity = InfoLogger::Severity::Debug;
281  level = 12;
282  } else if (metadata.severity_name == fair::Logger::SeverityName(fair::Severity::debug2)) {
283  severity = InfoLogger::Severity::Debug;
284  level = 13;
285  } else if (metadata.severity_name == fair::Logger::SeverityName(fair::Severity::debug3)) {
286  severity = InfoLogger::Severity::Debug;
287  level = 14;
288  } else if (metadata.severity_name == fair::Logger::SeverityName(fair::Severity::debug4)) {
289  severity = InfoLogger::Severity::Debug;
290  level = 15;
291  } else if (metadata.severity_name == fair::Logger::SeverityName(fair::Severity::trace)) {
292  severity = InfoLogger::Severity::Debug;
293  level = 50;
294  }
295 
296  InfoLogger::InfoLoggerMessageOption opt = {
297  severity,
298  level,
299  InfoLogger::undefinedMessageOption.errorCode,
300  metadata.file.c_str(),
301  atoi(metadata.line.c_str())};
302 
303  if (logger) {
304  logger->log(opt, *ctx, "%s", content.c_str());
305  }
306  };
307 };

312 o2::framework::ServiceSpec CommonServices::infologgerSpec()
313 {
314  return ServiceSpec{
315  .name = "infologger",
316  .init = [](ServiceRegistry& services, DeviceState&, fair::mq::ProgOptions& options) -> ServiceHandle {
317  auto infoLoggerMode = options.GetPropertyAsString("infologger-mode");
318  auto infoLoggerSeverity = options.GetPropertyAsString("infologger-severity");
319  if (infoLoggerSeverity.empty() == false && options.GetPropertyAsString("infologger-mode") == "") {
320  LOGP(info, "Using O2_INFOLOGGER_MODE=infoLoggerD since infologger-severity is set");
321  infoLoggerMode = "infoLoggerD";
322  }
323  if (infoLoggerMode != "") {
324  setenv("O2_INFOLOGGER_MODE", infoLoggerMode.c_str(), 1);
325  }
326  char const* infoLoggerEnv = getenv("O2_INFOLOGGER_MODE");
327  if (infoLoggerEnv == nullptr || strcmp(infoLoggerEnv, "none") == 0) {
328  return ServiceHandle{.hash = TypeIdHelpers::uniqueId<MissingService>(),
329  .instance = nullptr,
330  .kind = ServiceKind::Serial,
331  .name = "infologger"};
332  }
333  InfoLogger* infoLoggerService = nullptr;
334  try {
335  infoLoggerService = new InfoLogger;
336  } catch (...) {
337  LOGP(error, "Unable to initialise InfoLogger with O2_INFOLOGGER_MODE={}.", infoLoggerMode);
338  return ServiceHandle{.hash = TypeIdHelpers::uniqueId<MissingService>(),
339  .instance = nullptr,
340  .kind = ServiceKind::Serial,
341  .name = "infologger"};
342  }
343  auto infoLoggerContext = &services.get<InfoLoggerContext>();
344  // Only print the first 10 characters and the last 18 if the
345  // string length is greater than 32 bytes.
346  auto truncate = [](std::string in) -> std::string {
347  if (in.size() < 32) {
348  return in;
349  }
350  char name[32];
351  memcpy(name, in.data(), 10);
352  name[10] = '.';
353  name[11] = '.';
354  name[12] = '.';
355  memcpy(name + 13, in.data() + in.size() - 18, 18);
356  name[31] = 0;
357  return name;
358  };
359  infoLoggerContext->setField(InfoLoggerContext::FieldName::Facility, truncate(services.get<DeviceSpec const>().name));
360  infoLoggerContext->setField(InfoLoggerContext::FieldName::System, std::string("DPL"));
361  infoLoggerService->setContext(*infoLoggerContext);
362 
363  if (infoLoggerSeverity != "") {
364  fair::Logger::AddCustomSink("infologger", infoLoggerSeverity, createInfoLoggerSinkHelper(infoLoggerService, infoLoggerContext));
365  }
366  return ServiceHandle{.hash = TypeIdHelpers::uniqueId<InfoLogger>(),
367  .instance = infoLoggerService,
368  .kind = ServiceKind::Serial,
369  .name = "infologger"};
370  },
371  .configure = noConfiguration(),
372  .kind = ServiceKind::Serial};
373 }

528 // CCDB Support service
529 o2::framework::ServiceSpec CommonServices::ccdbSupportSpec()
530 {
531  return ServiceSpec{
532  .name = "ccdb-support",
533  .init = [](ServiceRegistry& services, DeviceState&, fair::mq::ProgOptions&) -> ServiceHandle {
534  // iterate on all the outputs matchers
535  auto& spec = services.get<DeviceSpec const>();
536  for (auto& output : spec.outputs) {
537  if (DataSpecUtils::match(output.matcher, ConcreteDataTypeMatcher{"FLP", "DISTSUBTIMEFRAME"})) {
538  LOGP(debug, "Optional inputs support enabled");
539  return ServiceHandle{.hash = TypeIdHelpers::uniqueId<CCDBSupport>(), .instance = new CCDBSupport, .kind = ServiceKind::Serial};
540  }
541  }
542  return ServiceHandle{.hash = TypeIdHelpers::uniqueId<CCDBSupport>(), .instance = nullptr, .kind = ServiceKind::Serial};
543  },
544  .configure = noConfiguration(),
545  .postProcessing = [](ProcessingContext& pc, void* service) {
546  if (!service) {
547  return;
548  }
549  if (pc.services().get<DeviceState>().streaming == StreamingState::EndOfStreaming) {
550  if (pc.outputs().countDeviceOutputs(true) == 0) {
551  LOGP(debug, "We are in EoS w/o outputs, do not automatically add DISTSUBTIMEFRAME to outgoing messages");
552  return;
553  }
554  }
555  const auto ref = pc.inputs().getFirstValid(true);
556  const auto* dh = DataRefUtils::getHeader<o2::header::DataHeader*>(ref);
557  const auto* dph = DataRefUtils::getHeader<DataProcessingHeader*>(ref);
558 
559  // For any output that is a FLP/DISTSUBTIMEFRAME with subspec != 0,
560  // we create a new message.
561  InputSpec matcher{"matcher", ConcreteDataTypeMatcher{"FLP", "DISTSUBTIMEFRAME"}};
562  for (auto& output : pc.services().get<DeviceSpec const>().outputs) {
563  if ((output.timeslice % output.maxTimeslices) != 0) {
564  continue;
565  }

580 // Decongestion service
581 // If we do not have any Timeframe input, it means we must be creating timeslices
582 // in order and that we should propagate the oldest possible timeslice at the end
583 // of each processing step.
584 o2::framework::ServiceSpec CommonServices::decongestionSpec()
585 {
586  return ServiceSpec{
587  .name = "decongestion",
588  .init = [](ServiceRegistry& services, DeviceState&, fair::mq::ProgOptions& options) -> ServiceHandle {
589  DecongestionService* decongestion = new DecongestionService();
590  for (auto& input : services.get<DeviceSpec const>().inputs) {
591  if (input.matcher.lifetime == Lifetime::Timeframe) {
592  LOGP(detail, "Found a Timeframe input, we cannot update the oldest possible timeslice");
593  decongestion->isFirstInTopology = false;
594  break;
595  }
596  }
597  return ServiceHandle{TypeIdHelpers::uniqueId<DecongestionService>(), decongestion, ServiceKind::Serial};
598  },
599  .postForwarding = [](ProcessingContext& ctx, void* service) {
600  DecongestionService* decongestion = reinterpret_cast<DecongestionService*>(service);
601  if (decongestion->isFirstInTopology == false) {
602  LOGP(debug, "We are not the first in the topology, do not update the oldest possible timeslice");
603  return;
604  }
605  auto& timesliceIndex = ctx.services().get<TimesliceIndex>();
606  auto& relayer = ctx.services().get<DataRelayer>();
607  timesliceIndex.updateOldestPossibleOutput();
608  auto& proxy = ctx.services().get<FairMQDeviceProxy>();
609  auto oldestPossibleOutput = relayer.getOldestPossibleOutput();
610  if (oldestPossibleOutput.timeslice.value == decongestion->lastTimeslice) {
611  LOGP(debug, "Not sending already sent value");
612  return;
613  }
614  if (oldestPossibleOutput.timeslice.value < decongestion->lastTimeslice) {
615  LOGP(error, "We are trying to send a oldest possible timeslice {} that is older than the last one we already sent {}",
616  oldestPossibleOutput.timeslice.value, decongestion->lastTimeslice);
617  return;
618  }
619 
620  LOGP(debug, "Broadcasting possible output {} due to {} ({})", oldestPossibleOutput.timeslice.value,
621  oldestPossibleOutput.slot.index == -1 ? "channel" : "slot",
622  oldestPossibleOutput.slot.index == -1 ? oldestPossibleOutput.channel.value: oldestPossibleOutput.slot.index);
623  DataProcessingHelpers::broadcastOldestPossibleTimeslice(proxy, oldestPossibleOutput.timeslice.value);
624  DeviceSpec const& spec = ctx.services().get<DeviceSpec const>();
625  auto device = ctx.services().get<RawDeviceService>().device();
626  for (size_t fi = 0; fi < spec.forwards.size(); fi++) {
627  auto& channel = device->GetChannel(spec.forwards[fi].channel, 0);
628  // The oldest possible timeslice for a forwarded message
629  // is conservatively the one of the device doing the forwarding.
630  if (spec.forwards[fi].channel.rfind("from_", 0) == 0) {
631  auto oldestTimeslice = timesliceIndex.getOldestPossibleOutput();
632  DataProcessingHelpers::sendOldestPossibleTimeframe(channel, oldestTimeslice.timeslice.value);
633  LOGP(debug, "Forwarding to channel {} oldest possible timeslice {}", spec.forwards[fi].channel, oldestTimeslice.timeslice.value);
634  }
635  }
636  decongestion->lastTimeslice = oldestPossibleOutput.timeslice.value; },
637  .domainInfoUpdated = [](ServiceRegistry& services, size_t oldestPossibleTimeslice, ChannelIndex channel) {
638  DecongestionService& decongestion = services.get<DecongestionService>();
639  auto& relayer = services.get<DataRelayer>();
640  auto& timesliceIndex = services.get<TimesliceIndex>();
641  auto& proxy = services.get<FairMQDeviceProxy>();
642  LOGP(debug, "Received oldest possible timeframe {} from channel {}", oldestPossibleTimeslice, channel.value);
643  relayer.setOldestPossibleInput({oldestPossibleTimeslice}, channel);
644  timesliceIndex.updateOldestPossibleOutput();
645  auto oldestPossibleOutput = relayer.getOldestPossibleOutput();
646 
647  if (oldestPossibleOutput.timeslice.value == decongestion.lastTimeslice) {
648  LOGP(debug, "Not sending already sent value");
649  return;
650  }
651  if (oldestPossibleOutput.timeslice.value < decongestion.lastTimeslice) {
652  LOGP(error, "We are trying to send a timeslice {} that is older than the last one we sent {}",
653  oldestPossibleOutput.timeslice.value, decongestion.lastTimeslice);
654  return;
655  }
656  LOGP(debug, "Broadcasting possible output {}", oldestPossibleOutput.timeslice.value);
657  DataProcessingHelpers::broadcastOldestPossibleTimeslice(proxy, oldestPossibleOutput.timeslice.value);
658  DeviceSpec const& spec = services.get<DeviceSpec const>();
659  auto device = services.get<RawDeviceService>().device();
660  for (size_t fi = 0; fi < spec.forwards.size(); fi++) {
661  auto& channel = device->GetChannel(spec.forwards[fi].channel, 0);
662  // The oldest possible timeslice for a forwarded message
663  // is conservatively the one of the device doing the forwarding.
664  if (spec.forwards[fi].channel.rfind("from_", 0) == 0) {
665  auto oldestTimeslice = timesliceIndex.getOldestPossibleOutput();
666  LOGP(debug, "Forwarding to channel {} oldest possible timeslice {}", spec.forwards[fi].channel, oldestTimeslice.timeslice.value);
667  DataProcessingHelpers::sendOldestPossibleTimeframe(channel, oldestTimeslice.timeslice.value);
668  }
669  }
670  decongestion.lastTimeslice = oldestPossibleOutput.timeslice.value; },
671  .kind = ServiceKind::Serial};
672 }

703 auto sendRelayerMetrics(ServiceRegistry& registry, DataProcessingStats& stats) -> void
704 {
705  auto timeSinceLastUpdate = stats.beginIterationTimestamp - stats.lastSlowMetricSentTimestamp;
706  auto timeSinceLastLongUpdate = stats.beginIterationTimestamp - stats.lastVerySlowMetricSentTimestamp;
707  if (timeSinceLastUpdate < 5000) {
708  return;
709  }
710  // Derive the amount of shared memory used
711  auto& runningWorkflow = registry.get<RunningWorkflowInfo const>();
712  using namespace fair::mq::shmem;
713  auto& spec = registry.get<DeviceSpec const>();
714 
715  // FIXME: Ugly, but we do it only every 5 seconds...
716  if (spec.name == "readout-proxy") {
717  auto device = registry.get<RawDeviceService>().device();
718  long freeMemory = -1;
719  try {
720  freeMemory = Monitor::GetFreeMemory(ShmId{makeShmIdStr(device->fConfig->GetProperty<uint64_t>("shmid"))}, runningWorkflow.shmSegmentId);
721  } catch (...) {
722  }
723  if (freeMemory == -1) {
724  try {
725  freeMemory = Monitor::GetFreeMemory(SessionId{device->fConfig->GetProperty<std::string>("session")}, runningWorkflow.shmSegmentId);
726  } catch (...) {
727  }
728  }
729  if (freeMemory != -1) {
730  stats.availableManagedShm.store(freeMemory);
731  }
732  }
733 
734  auto performedComputationsSinceLastUpdate = stats.performedComputations - stats.lastReportedPerformedComputations;
735 
736  ZoneScopedN("send metrics");
737  auto& relayerStats = registry.get<DataRelayer>().getStats();
738  auto& monitoring = registry.get<Monitoring>();
739 
740  O2_SIGNPOST_START(MonitoringStatus::ID, MonitoringStatus::SEND, 0, 0, O2_SIGNPOST_BLUE);
741 
742  monitoring.send(Metric{(int)relayerStats.malformedInputs, "malformed_inputs"}.addTag(Key::Subsystem, Value::DPL));
743  monitoring.send(Metric{(int)relayerStats.droppedComputations, "dropped_computations"}.addTag(Key::Subsystem, Value::DPL));
744  monitoring.send(Metric{(int)relayerStats.droppedIncomingMessages, "dropped_incoming_messages"}.addTag(Key::Subsystem, Value::DPL));
745  monitoring.send(Metric{(int)relayerStats.relayedMessages, "relayed_messages"}.addTag(Key::Subsystem, Value::DPL));
746 
747  monitoring.send(Metric{(int)stats.errorCount, "errors"}.addTag(Key::Subsystem, Value::DPL));
748  monitoring.send(Metric{(int)stats.exceptionCount, "exceptions"}.addTag(Key::Subsystem, Value::DPL));
749  monitoring.send(Metric{(int)stats.pendingInputs, "inputs/relayed/pending"}.addTag(Key::Subsystem, Value::DPL));
750  monitoring.send(Metric{(int)stats.incomplete, "inputs/relayed/incomplete"}.addTag(Key::Subsystem, Value::DPL));
751  monitoring.send(Metric{(int)stats.inputParts, "inputs/relayed/total"}.addTag(Key::Subsystem, Value::DPL));
752  monitoring.send(Metric{stats.lastElapsedTimeMs, "elapsed_time_ms"}.addTag(Key::Subsystem, Value::DPL));
753  monitoring.send(Metric{stats.lastProcessedSize, "last_processed_input_size_byte"}
754  .addTag(Key::Subsystem, Value::DPL));
755  monitoring.send(Metric{stats.totalProcessedSize, "total_processed_input_size_byte"}
756  .addTag(Key::Subsystem, Value::DPL));
757  monitoring.send(Metric{stats.totalSigusr1.load(), "total_sigusr1"}.addTag(Key::Subsystem, Value::DPL));
758  monitoring.send(Metric{(stats.lastProcessedSize.load() / (stats.lastElapsedTimeMs.load() ? stats.lastElapsedTimeMs.load() : 1) / 1000),
759  "processing_rate_mb_s"}
760  .addTag(Key::Subsystem, Value::DPL));
761  monitoring.send(Metric{stats.lastLatency.minLatency, "min_input_latency_ms"}
762  .addTag(Key::Subsystem, Value::DPL));
763  monitoring.send(Metric{stats.lastLatency.maxLatency, "max_input_latency_ms"}
764  .addTag(Key::Subsystem, Value::DPL));
765  monitoring.send(Metric{(stats.lastProcessedSize / (stats.lastLatency.maxLatency ? stats.lastLatency.maxLatency : 1) / 1000), "input_rate_mb_s"}
766  .addTag(Key::Subsystem, Value::DPL));
767  monitoring.send(Metric{((float)performedComputationsSinceLastUpdate / (float)timeSinceLastUpdate) * 1000, "processing_rate_hz"}.addTag(Key::Subsystem, Value::DPL));
768  monitoring.send(Metric{(uint64_t)stats.performedComputations, "performed_computations"}.addTag(Key::Subsystem, Value::DPL));
769 
770  if (stats.availableManagedShm) {
771  monitoring.send(Metric{(uint64_t)stats.availableManagedShm, fmt::format("available_managed_shm_{}", runningWorkflow.shmSegmentId)}.addTag(Key::Subsystem, Value::DPL));
772  }
773 
774  if (stats.consumedTimeframes) {
775  monitoring.send(Metric{(uint64_t)stats.consumedTimeframes, "consumed-timeframes"}.addTag(Key::Subsystem, Value::DPL));
776  }
777 
778  stats.lastSlowMetricSentTimestamp.store(stats.beginIterationTimestamp.load());
779  stats.lastReportedPerformedComputations.store(stats.performedComputations.load());
780  O2_SIGNPOST_END(MonitoringStatus::ID, MonitoringStatus::SEND, 0, 0, O2_SIGNPOST_BLUE);
781 
782  auto device = registry.get<RawDeviceService>().device();
783 
784  uint64_t lastTotalBytesIn = stats.totalBytesIn.exchange(0);
785  uint64_t lastTotalBytesOut = stats.totalBytesOut.exchange(0);
786  uint64_t totalBytesIn = 0;
787  uint64_t totalBytesOut = 0;
788 
789  for (auto& channel : device->fChannels) {
790  totalBytesIn += channel.second[0].GetBytesRx();
791  totalBytesOut += channel.second[0].GetBytesTx();
792  }
793 
794  monitoring.send(Metric{(float)(totalBytesOut - lastTotalBytesOut) / 1000000.f / (timeSinceLastUpdate / 1000.f), "total_rate_out_mb_s"}
795  .addTag(Key::Subsystem, Value::DPL));
796  monitoring.send(Metric{(float)(totalBytesIn - lastTotalBytesIn) / 1000000.f / (timeSinceLastUpdate / 1000.f), "total_rate_in_mb_s"}
797  .addTag(Key::Subsystem, Value::DPL));
798  stats.totalBytesIn.store(totalBytesIn);
799  stats.totalBytesOut.store(totalBytesOut);
800  // Things which we report every 30s
801  if (timeSinceLastLongUpdate < 30000) {
802  return;
803  }
804  stats.lastVerySlowMetricSentTimestamp.store(stats.beginIterationTimestamp.load());
805 };

808 auto flushMetrics(ServiceRegistry& registry, DataProcessingStats& stats) -> void
809 {
810  auto timeSinceLastUpdate = stats.beginIterationTimestamp - stats.lastMetricFlushedTimestamp;
811  static int counter = 0;
812  if (timeSinceLastUpdate < 1000) {
813  if (counter++ > 10) {
814  return;
815  }
816  } else {
817  counter = 0;
818  }
819 
820  ZoneScopedN("flush metrics");
821  auto& monitoring = registry.get<Monitoring>();
822  auto& relayer = registry.get<DataRelayer>();
823 
824  O2_SIGNPOST_START(MonitoringStatus::ID, MonitoringStatus::FLUSH, 0, 0, O2_SIGNPOST_RED);
825  // Send all the relevant metrics for the relayer to update the GUI
826  // FIXME: do a delta with the previous version if too many metrics are still
827  // sent...
828  for (size_t si = 0; si < stats.statesSize.load(); ++si) {
829  auto value = std::atomic_load_explicit(&stats.relayerState[si], std::memory_order_relaxed);
830  std::atomic_thread_fence(std::memory_order_acquire);
831  monitoring.send({value, fmt::format("data_relayer/{}", si), o2::monitoring::Verbosity::Debug});
832  }
833  relayer.sendContextState();
834  monitoring.flushBuffer();
835  stats.lastMetricFlushedTimestamp.store(stats.beginIterationTimestamp.load());
836  O2_SIGNPOST_END(MonitoringStatus::ID, MonitoringStatus::FLUSH, 0, 0, O2_SIGNPOST_RED);
837 };
