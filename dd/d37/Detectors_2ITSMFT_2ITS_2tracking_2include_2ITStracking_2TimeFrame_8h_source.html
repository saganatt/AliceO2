225 inline gsl::span<const Vertex> TimeFrame::getPrimaryVertices(int rof) const
226 {
227  const int start = mROframesPV[rof];
228  const int stop_idx = rof >= mNrof - 1 ? mNrof : rof + 1;
229  int delta = mMultiplicityCutMask[rof] ? mROframesPV[stop_idx] - start : 0; // return empty span if Rof is excluded
230  return {&mPrimaryVertices[start], static_cast<gsl::span<const Vertex>::size_type>(delta)};
231 }

234 {
235  return {&mPrimaryVertices[mROframesPV[romin]], static_cast<gsl::span<const Vertex>::size_type>(mROframesPV[romax + 1] - mROframesPV[romin])};
236 }
