77 void CosmicsMatchingSpec::init(InitContext& ic)
78 {
79  mTimer.Stop();
80  mTimer.Reset();
81  //-------- init geometry and field --------//

83  o2::base::Propagator::initFieldFromGRP();
84  o2::its::GeometryTGeo::Instance()->fillMatrixCache(o2::math_utils::bit2Mask(o2::math_utils::TransformType::T2GRot) | o2::math_utils::bit2Mask(o2::math_utils::TransformType::T2L));
85 
86  // this is a hack to provide Mat.LUT from the local file, in general will be provided by the framework from CCDB
87  std::string matLUTPath = ic.options().get<std::string>("material-lut-path");
88  std::string matLUTFile = o2::base::NameConf::getMatLUTFileName(matLUTPath);
89  if (o2::utils::Str::pathExists(matLUTFile)) {
90  auto* lut = o2::base::MatLayerCylSet::loadFromFile(matLUTFile);
91  o2::base::Propagator::Instance()->setMatLUT(lut);
92  LOG(info) << "Loaded material LUT from " << matLUTFile;
93  } else {
94  LOG(info) << "Material LUT " << matLUTFile << " file is absent, only TGeo can be used";
95  }
96 
97  mMatching.setDebugFlag(ic.options().get<int>("debug-tree-flags"));
98  mMatching.setUseMC(mUseMC);
99  //

102 void CosmicsMatchingSpec::run(ProcessingContext& pc)
103 {
104  mTimer.Start(false);
105  RecoContainer recoData;
106  recoData.collectData(pc, *mDataRequest.get());
107  updateTimeDependentParams(pc); // Make sure this is called after recoData.collectData, which may load some conditions
108 
109  mMatching.process(recoData);
110  pc.outputs().snapshot(Output{"GLO", "COSMICTRC", 0, Lifetime::Timeframe}, mMatching.getCosmicTracks());
111  if (mUseMC) {
112  pc.outputs().snapshot(Output{"GLO", "COSMICTRC_MC", 0, Lifetime::Timeframe}, mMatching.getCosmicTracksLbl());
113  }
114  mTimer.Stop();
115 }

117 void CosmicsMatchingSpec::updateTimeDependentParams(ProcessingContext& pc)
118 {
119  static bool initOnceDone = false;
120  if (!initOnceDone) { // this params need to be queried only once
121  initOnceDone = true;
122 
123  // pc.inputs().get<o2::itsmft::TopologyDictionary*>("cldict"); // called by the RecoContainer
124  // also alpParams is called by the RecoContainer
125  std::unique_ptr<o2::parameters::GRPObject> grp{o2::parameters::GRPObject::loadFrom()};
126  const auto& alpParams = o2::itsmft::DPLAlpideParam<o2::detectors::DetID::ITS>::Instance();
127  if (!grp->isDetContinuousReadOut(DetID::ITS)) {
128  mMatching.setITSROFrameLengthMUS(alpParams.roFrameLengthTrig / 1.e3); // ITS ROFrame duration in \mus
129  } else {
130  mMatching.setITSROFrameLengthMUS(alpParams.roFrameLengthInBC * o2::constants::lhc::LHCBunchSpacingNS * 1e-3); // ITS ROFrame duration in \mus
131  }
132 
133  mMatching.init();
134  }

152 DataProcessorSpec getCosmicsMatchingSpec(GTrackID::mask_t src, bool useMC)
153 {
154  std::vector<OutputSpec> outputs;
155  auto dataRequest = std::make_shared<DataRequest>();
156 
157  dataRequest->requestTracks(src, useMC);
158  dataRequest->requestClusters(src, false); // no MC labels for clusters needed for refit only
159 
160  outputs.emplace_back("GLO", "COSMICTRC", 0, Lifetime::Timeframe);
161  if (useMC) {
162  outputs.emplace_back("GLO", "COSMICTRC_MC", 0, Lifetime::Timeframe);
163  }
164 
165  return DataProcessorSpec{
166  "cosmics-matcher",
167  dataRequest->inputs,
168  outputs,
169  AlgorithmSpec{adaptFromTask<CosmicsMatchingSpec>(dataRequest, useMC)},
170  Options{
171  {"material-lut-path", VariantType::String, "", {"Path of the material LUT file"}},
172  {"debug-tree-flags", VariantType::Int, 0, {"DebugFlagTypes bit-pattern for debug tree"}}}};
173 }
